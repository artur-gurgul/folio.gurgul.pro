\sheettitle{macOS keychains \& the \ct{security} CLI (CI signing)}{security-build · folio}

\oneliner{macOS has \textbf{two keychains}: the legacy \textbf{file-based} ones
(\ct{login.keychain-db}, \ct{System.keychain}, any file you create) searched in a \textbf{search
list} and guarded by per-item \textbf{ACLs + a partition list}, and the iOS-style \textbf{data
protection keychain} (one per user, guarded by \textbf{access groups} from entitlements; needed for
iCloud Keychain, biometrics, Secure Enclave). \ct{codesign} finds its \textbf{identity} (certificate +
private key) in the \emph{file-based} search list — so CI must create, unlock and authorise a keychain.}

\hd{How it works}
\begin{itemize}
  \item \textbf{File-based} = classic Mac OS heritage, ``on the road to deprecation''
        (\ct{SecKeychainCreate} deprecated, macOS 12 SDK); \textbf{data protection} = iOS's, on
        the Mac since iCloud Keychain (10.9) — opt in with \ct{kSecUseDataProtectionKeychain}.
  \item \textbf{Item classes} (\ct{security} 4-letter codes): \ct{genp} generic password,
        \ct{inet} internet password, \ct{cert}, \ct{keys}; \ct{idnt} = identity, assembled at
        query time from a cert whose public key matches a stored private key.
  \item \textbf{ACL} = which apps may use an item without a prompt (\ct{-T /usr/bin/codesign};
        \ct{-A} = any app, no prompt). Code identity is the app's \textbf{designated
        requirement}, so a re-signed or re-channelled app (e.g. TestFlight build) gets a
        new prompt.
  \item \textbf{Partition list} (added later to the file-based keychain, no public API): an
        extra ACL tag on a key naming allowed \emph{signer groups} — \ct{apple:} (Apple-signed
        system code), \ct{apple-tool:} (Apple tools such as \ct{codesign}/\ct{security}),
        \ct{teamid:ABCDE12345}. A key imported by \ct{security import} gets only
        \ct{apple-tool:}\unverified.
  \item \textbf{Identity}: the \ct{.cer} Apple issues holds the public key only; the private
        key was created on the Mac that made the CSR. Export both as a \ct{.p12}.
\end{itemize}

\hd{File-based vs data protection keychain}
{\footnotesize
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{13mm}>{\raggedright\arraybackslash}p{32mm}>{\raggedright\arraybackslash}p{28mm}@{}}
\toprule
& \textbf{file-based} & \textbf{data protection} \\ \midrule
where & files (login, System, yours) in a \textbf{search list} & one per user, by caller context \\
who may read & per-item \textbf{ACL} + partition list; prompts & \textbf{access groups} (entitlements) + \ct{SecAccessControl} \\

daemons / root & yes (System keychain) & no — user login context only \\

\end{tabular}}

\hd{Remember}
\textbf{``Search list finds it, unlock opens it, ACL + partition list let \ct{codesign} touch it.''}

\hd{Example — the CI keychain recipe}
\begin{lstlisting}[language=ShSheet]
KC="$RUNNER_TEMP/build.keychain-db"; PW="$(uuidgen)"   # throwaway
security create-keychain -p "$PW" "$KC"
security set-keychain-settings -lut 21600 "$KC"  # lock on sleep, after 6 h
security unlock-keychain -p "$PW" "$KC"
security import dist.p12 -k "$KC" -P "$P12_PW" -f pkcs12 \
         -T /usr/bin/codesign -T /usr/bin/security   # ACL: these tools
security set-key-partition-list -S apple-tool:,apple: -s -k "$PW" "$KC"
security list-keychains -d user -s "$KC" login.keychain-db  # keep login!
security find-identity -v -p codesigning "$KC"   # expect 1 valid identity
xcodebuild archive ... OTHER_CODE_SIGN_FLAGS="--keychain $KC"
security delete-keychain "$KC"                   # always, in a trap/post step
\end{lstlisting}

\columnbreak

\hd{Where the pieces come from}
\begin{itemize}
  \item \textbf{Apple WWDR intermediate} must be in a searched keychain or \ct{codesign} cannot
        build the chain — download it from Apple PKI (apple.com/certificateauthority).
  \item \textbf{Keychain Access} edits ACLs (``Access Control'' tab) of \emph{file-based} items
        only; \ct{security dump-keychain -a} prints the ACLs.
\end{itemize}

\hd{\ct{security} — the commands you are asked about}
{\footnotesize
\begin{tabular}{@{}>{\ttfamily\raggedright\arraybackslash}p{36mm}>{\raggedright\arraybackslash}p{38mm}@{}}
\toprule
create-keychain -p & new file keychain (\ct{.keychain-db}) + password \\
set-keychain-settings -lut & \ct{-l} lock on sleep · \ct{-u} lock after \ct{-t} secs; no flags = never lock \\
unlock-keychain -p & gate A; again in every new session (SSH) \\
import -k -P -T/-A -f & \ct{.p12}/\ct{.cer} into keychain \ct{-k}; \ct{-T} app into the key's ACL \\
set-key-partition-list -S -s -k & partitions · signing keys only · keychain password \\
list-keychains -d user -s & \textbf{set} the search list (no \ct{-s} = print it) \\
default-keychain -s & where new items land; CI rarely needs it \\
find-identity -v -p codesigning & valid signing identities: SHA-1 + name \\
find-generic-password -s -a -w & read a secret (\ct{-w} prints only it) \\
add-generic-password -U & store / update (\ct{-U}) a secret \\
add-trusted-cert -r trustRoot & trust a private CA (\ct{-d}: admin domain) \\
cms -D -i & decode CMS, e.g. a \ct{.mobileprovision} \\
delete-keychain & remove the file + its search-list entry \\
\bottomrule
\end{tabular}}

\hd{Traps}
\begin{itemize}
  \trap{Fixing the ACL won't help a \textbf{locked} keychain — unlock first (Apple DTS).}
  \trap{\ct{list-keychains -s \$KC} alone \textbf{drops login} from the search list.}
  \trap{\ct{find-identity}: ``0 valid identities'' though the cert is there $\to$ private key
        missing (imported \ct{.cer}, not \ct{.p12}) or WWDR intermediate missing/expired.}
  \trap{\ct{sudo codesign}: root can't see your login keychain.}
  \trap{\ct{-A} on a long-lived keychain = every app reads the key silently; OK only on a
        throwaway CI keychain.}
  \trap{Mac \ct{SecItem} \emph{without} \ct{kSecUseDataProtectionKeychain} lands in the
        \textbf{file-based} keychain — ACL prompts after re-signing, no access-group sharing.}
  \trap{\ct{security find-generic-password -w} in a CI log prints the secret.}
\end{itemize}
