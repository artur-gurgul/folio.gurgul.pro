\sheettitle{Notarization \& distribution channels}{security-build · folio}

\oneliner{The \textbf{channel} decides who may install and who vouches: App Store, TestFlight,
Custom and Unlisted apps go through \textbf{App Review} and are \textbf{re-signed by Apple}; Ad Hoc
and Enterprise run on \textbf{your} signature + profile; a Mac app outside the store is signed
with \textbf{Developer ID}, uses the \textbf{hardened runtime} and is \textbf{notarized} — an
automated malware + signing scan whose \textbf{ticket} Gatekeeper checks at first launch.
``Notarization'' for \emph{iOS} (EU, via alternative marketplaces or the web) is a different thing: a
baseline App Review.}

\hd{How macOS notarization works}
\begin{itemize}
  \item \textbf{Requirements}: every executable signed with \textbf{Developer ID} (not
        Development/Distribution/ad hoc) + \textbf{hardened runtime} + \textbf{secure
        timestamp}; no \ct{get-task-allow}; SDK $\geq$ 10.9; ASCII-XML entitlements.
        \textbf{Mandatory} since macOS 10.15 (software built after 1 June 2019).
  \item \textbf{Hardened runtime} blocks JIT, \ct{DYLD\_*} env vars, foreign dylibs, debugger
        attach — each re-enabled only by a \ct{com.apple.security.cs.*} entitlement
        (\ct{allow-jit}, \ct{disable-library-validation}…).
  \item \textbf{What is scanned}: malicious content + code-signing problems; an audit trail per
        signing key is kept.
  \item \textbf{Tools}: \ct{notarytool} (Xcode 13+). Apple stopped accepting \ct{altool} and
        Xcode 13- uploads on \textbf{1 Nov 2023}. Auth: keychain profile (Apple ID +
        app-specific password + Team ID) or an App Store Connect API key (\ct{\dd key \dd key-id
        \dd issuer}).
  \item \textbf{Packaging}: notarize the thing you ship; a \ct{.zip} can't be stapled (staple the
        app, then re-zip). DMG: sign the app, build the DMG, sign the DMG, notarize + staple the
        DMG. Installer: \ct{pkgbuild} (component) / \ct{productbuild} (product archive), signed
        with \textbf{Developer ID Installer}; custom installers = two rounds (payload, then
        installer).
\end{itemize}

\hd{iOS outside the App Store — EU (and Japan)}
\begin{itemize}
  \item Reviewed under the \textbf{Notarization Review Guidelines} only (security, privacy,
        device integrity) $\to$ alternative channels only; full App Review $\to$ both.
  \item Apple \textbf{encrypts and signs} every alternatively distributed app; install checks
        for tampering and an authorised source; known malware $\to$ won't launch, installs blocked.
  \item Marketplaces need Apple's authorisation (USD 1 M letter of credit). Japan: iOS
        Notarization too.
\end{itemize}

\hd{Remember}
\textbf{``Sign inside-out with a timestamp and the runtime, notarize the outermost thing,
staple what you ship.''}

\hd{Example — Developer ID release of a DMG}
\begin{lstlisting}[language=ShSheet]
ID="Developer ID Application: Alice (ABCDE12345)"
codesign -s "$ID" -o runtime --timestamp -f App.app   # after nested code
hdiutil create -volname App -srcfolder App.app -ov -format UDZO App.dmg
codesign -s "$ID" --timestamp App.dmg
xcrun notarytool store-credentials AC --apple-id alice@example.com \
      --team-id ABCDE12345            # prompts for an app-specific password
xcrun notarytool submit App.dmg --keychain-profile AC --wait
xcrun notarytool log <submission-id> --keychain-profile AC  # if Invalid
xcrun stapler staple App.dmg && xcrun stapler validate App.dmg
spctl -a -vvv -t exec App.app     # accepted, source=Notarized Developer ID
productbuild --component App.app /Applications \
      --sign "Developer ID Installer: Alice (ABCDE12345)" App.pkg
\end{lstlisting}

\hd{Certificate facts that decide channels}
\begin{itemize}
  \item One certificate \emph{of each distribution type} per team — except Developer ID; only
        the Account Holder or an Admin can create distribution certificates.
  \item Developer ID \textbf{Application} expired: existing builds still install and run;
        \textbf{Installer} expired: pkgs install only if timestamped; \textbf{revoked}: no installs.
\end{itemize}

\hd{Traps}
\begin{itemize}
  \trap{\textbf{Notarized $\neq$ reviewed}: macOS notarization is automated malware/signing
        scanning; App Review is a different process.}
  \trap{Signing a DMG with \emph{Developer ID Installer}, or a pkg with \emph{Application} — each
        container has its own certificate type.}
  \trap{\ct{codesign \dd deep} ``fixes'' the notary's nested-code complaint but stamps the app's
        entitlements on every helper — sign each item.}
  \trap{Missing \ct{\dd timestamp}: the notary rejects it (``does not include a secure
        timestamp''); for a pkg the timestamp is what keeps it installable after the
        Installer cert expires.}
  \trap{Enterprise cert for customers/betas — revoked $\to$ every copy dies; use Custom apps.}
\end{itemize}
