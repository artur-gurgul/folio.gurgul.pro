\sheettitle{The modern Jetpack app stack}{cross-platform · folio}

\oneliner{Google's recommended app: \textbf{UI} (Compose) observes \texttt{StateFlow} from a
\textbf{ViewModel}, which calls a \textbf{repository} whose \textbf{single source of truth is
the local DB} (Room) — the network only \emph{writes} into it. \textbf{Hilt} builds the graph
at \textbf{compile time}; \textbf{WorkManager} runs deferrable work that must survive process
death and reboot. Every piece has an iOS twin (green).}

\section{Hilt (Dagger underneath)}
\begin{itemize}\raggedright
  \item \texttt{@HiltAndroidApp} Application $\to$ \texttt{@AndroidEntryPoint} Activity/Service
        $\to$ \texttt{@HiltViewModel} + \texttt{@Inject constructor}. A missing binding
        \textbf{fails the build} (Koin fails at runtime). Use \textbf{KSP}, not kapt.
  \item \texttt{@Module @InstallIn(X::class)}: \texttt{@Binds} = interface $\to$ impl, no code;
        \texttt{@Provides} = factory for types you don't own (Retrofit, Room). Two of one
        type: \texttt{@Qualifier}. Runtime args: \texttt{@AssistedInject}; tests:
        \texttt{@TestInstallIn} swaps a module.
\end{itemize}

\section{ViewModel scope + SavedStateHandle}
\raggedright Each \textbf{nav back-stack entry} is a \texttt{ViewModelStore}: one VM per destination,
cleared when popped; share one across a flow by scoping it to the graph's entry.
\texttt{SavedStateHandle} = Bundle-backed map that survives \textbf{process death}
(\texttt{getStateFlow(k, d)}, \texttt{toRoute<T>()}) — ids, not data.

\section{Room}
\begin{itemize}\raggedright
  \item SQL \textbf{verified at compile time}. A \texttt{Flow} query re-emits on \emph{any}
        write to its \textbf{table} — \texttt{distinctUntilChanged()}. \texttt{@Transaction}
        for multi-query reads (\texttt{@Relation}); \texttt{withTransaction\{\}} for writes.
  \item Migrations: \texttt{Migration(1, 2) \{ execSQL(…) \}} or \texttt{@AutoMigration}
        (needs the exported schema JSON in git) + \texttt{MigrationTestHelper};
        \texttt{fallbackToDestructiveMigration()} = data loss. KMP since \textbf{2.7}.
\end{itemize}

\section{Example — the stack in 13 lines}
\begin{lstlisting}[language=KotlinSheet]
@Dao interface NoteDao {
  @Query("SELECT * FROM Note") fun all(): Flow<List<Note>>
  @Upsert suspend fun upsert(n: List<Note>) }
@Singleton class NoteRepo @Inject constructor(
    private val dao: NoteDao, private val api: NoteApi) {
  val notes = dao.all()                           // UI reads the DB
  suspend fun refresh() = dao.upsert(api.fetch()) // network -> DB
}
@HiltViewModel class NotesVM @Inject constructor(
    saved: SavedStateHandle, repo: NoteRepo) : ViewModel() {
  val folder = saved.toRoute<Notes>().folder      // typed nav arg
  val ui = repo.notes.stateIn(viewModelScope,
      SharingStarted.WhileSubscribed(5_000), emptyList()) }
\end{lstlisting}

\section{WorkManager — guaranteed, not punctual}
\begin{itemize}\raggedright
  \item Persisted in its own SQLite DB: survives death and \textbf{reboot}, honours Doze.
        \texttt{CoroutineWorker} $\to$ \texttt{Result.success/failure/retry} (backoff
        exponential, 30\,s).
  \item Constraints (network, charging, battery/storage not low, idle); periodic
        $\geq$ \textbf{15 min}. \texttt{enqueueUniqueWork(name, KEEP/REPLACE/APPEND, r)}
        stops duplicate syncs; \texttt{beginWith(a).then(b)} chains.
  \item \textbf{Expedited} (\texttt{setExpedited(OutOfQuotaPolicy.…)}): quota by standby
        bucket; below Android 12 runs as a foreground service (\texttt{getForegroundInfo}).
        $>$10 min: \texttt{setForeground()}. Not for exact alarms or ``now, visibly''.
\end{itemize}

\section{DataStore (replaces SharedPreferences)}
\raggedright\textbf{Preferences} (keys) or \textbf{typed} (Proto / JSON + \texttt{Serializer<T>}); reads
\texttt{Flow<T>}, writes \texttt{edit\{\}}/\texttt{updateData\{\}} — suspend,
\textbf{transactional}. SharedPreferences: sync reads on main, \texttt{apply()} blocks in
\texttt{fsync}. \textbf{One instance per file per process} or \texttt{IllegalStateException}.

\section{Navigation \& Paging}
\begin{itemize}\raggedright
  \item \textbf{Nav Compose 2.8+}: \texttt{@Serializable} routes, \texttt{composable<Profile>},
        \texttt{navigate(Profile(id))}, \texttt{toRoute()} — no string routes.
        \textbf{Navigation 3} (1.0 Nov 2025): \textbf{you own the back stack}
        (\texttt{SnapshotStateList<NavKey>}), \texttt{NavDisplay} + \texttt{entryProvider}
        — like \texttt{NavigationStack(path:)}.
  \item \textbf{Paging 3}: \texttt{PagingSource.load()} $\to$ \texttt{LoadResult.Page(data,
        prevKey, nextKey)}; \texttt{Pager.flow.cachedIn(scope)}; \texttt{RemoteMediator}
        fills Room from the network.
\end{itemize}

\section{Traps}
\begin{itemize}\raggedright
  \trap{\texttt{@Singleton} everywhere — state leaks across logins.}
  \trap{``WorkManager runs it now'' — \emph{eventually}, once constraints hold.}
  \trap{A DataStore created in a composable — two instances, crash.}
\end{itemize}

\textbf{Remember:} \emph{UI $\leftarrow$ VM $\leftarrow$ repo $\leftarrow$ DB; the network
writes the DB · Hilt at compile time · WorkManager survives reboot.}
