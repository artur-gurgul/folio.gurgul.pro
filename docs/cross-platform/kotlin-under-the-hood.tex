\sheettitle{Kotlin under the hood — what reaches the JVM}{cross-platform · folio}

\oneliner{The JVM knows nothing about Kotlin: no top-level functions, no properties, no
companions, no default arguments, no value classes, no \texttt{suspend}. \textbf{kotlinc}
lowers each feature to plain classes, static methods, synthetic bridges and runtime helper
calls (\texttt{kotlin.jvm.internal.Intrinsics}). Knowing the lowering explains Java-interop
annotations, boxing costs, \texttt{\$default} frames in stack traces, and why a read-only
\texttt{List} can still change.}

\section{Declarations $\to$ JVM}
\begin{itemize}\raggedright
  \item \textbf{Properties} = private field + \texttt{getX()}/\texttt{setX()} (\texttt{isX}
        stays \texttt{isX}). \texttt{@JvmField} exposes the field; \texttt{const val}
        (primitive/String, top-level/\texttt{object}/companion) is a compile-time constant
        \textbf{copied into callers} — change it, recompile them.
  \item \textbf{Companion} = nested class \texttt{Outer\$Companion} + static field; its
        members are \emph{instance} methods. \texttt{@JvmStatic} adds a real static in
        \texttt{Outer}, so Java writes \texttt{Api.create()}.
  \item \textbf{Default args}: one body + synthetic \texttt{f\$default(args…, int mask,
        Object)}; each bit of \texttt{mask} says ``use the default''. Java sees only the full
        signature — add \texttt{@JvmOverloads} for telescoping overloads.
  \item \textbf{Lambdas}: \texttt{invokedynamic} + \texttt{LambdaMetafactory} by default
        since \textbf{2.0} (smaller APKs; not serializable; \texttt{-Xlambdas=class} reverts).
        Captured \texttt{var}s become \texttt{Ref.IntRef}/\texttt{ObjectRef}.
  \item \textbf{\texttt{lateinit var}}: non-null field; the getter null-checks and throws
        \texttt{UninitializedPropertyAccessException}; \cc\texttt{x.isInitialized}; no
        primitives. A nullable \texttt{var} instead makes every read \sq/\bb.
  \item No checked exceptions in Kotlin: without \texttt{@Throws(IOException}\cc\texttt{class)} Java
        cannot \texttt{catch (IOException e)} around the call — ``never thrown in try''.
\end{itemize}

\section{Example — Java-facing API}
\begin{lstlisting}[language=KotlinSheet]

  val url: String, val retries: Int = 3, val gzip: Boolean = true) {
  companion object {
    const val VERSION = 2        // Java: Client.VERSION (inlined)
    @JvmField val TAG = "client" // Java: Client.TAG (static field)
    @JvmStatic fun local() = Client("https://example.com")
    fun plain() = 1              // Java: Client.Companion.plain()
  }

}

\section{Control flow \& collections}
\begin{itemize}\raggedright
  \item \texttt{when (s)} on \textbf{sealed}/\texttt{is}: an \texttt{instanceof} chain; on
        JVM 21+ a single \texttt{invokedynamic} \texttt{SwitchBootstraps.typeSwitch}
        (experimental 2.2.20, \textbf{default since 2.4.20}). On an \textbf{enum}: synthetic
        \texttt{\$WhenMappings} ordinal table + \texttt{tableswitch}\unverified.
  \item \textbf{Read-only $\neq$ immutable}: \texttt{List} is a compile-time view of
        \texttt{java.util.List}; \texttt{listOf(a, b)} is \texttt{Arrays.asList}-backed\unverified,
        \texttt{mutableListOf} an \texttt{ArrayList} — a cast or Java code can mutate it.
        \texttt{setOf} iterates in insertion order.
  \item \textbf{Iterable} ops are eager — every step builds a new list. \textbf{Sequence}
        is lazy, \emph{per element} (map 1 $\to$ filter 1 $\to$ \ldots), short-circuits
        (\texttt{first}); pays off for long chains/big data, not 10 items or \texttt{sorted()}.
\end{itemize}

\section{Boxing}
\begin{itemize}\raggedright
  \item \texttt{Int} $\to$ \texttt{int}; \texttt{Int?}, \texttt{List<Int>}, generic
        \texttt{T} $\to$ \texttt{Integer}. \texttt{IntArray} = \texttt{int[]},
        \texttt{Array<Int>} = \texttt{Integer[]}. \eqeqeq{} on boxes compares objects.
\end{itemize}

\section{Compiler \& targets}
\begin{itemize}\raggedright
  \item \textbf{K2} (stable, default since \textbf{2.0}, May 2024): new frontend \textbf{FIR}
        feeds a shared \textbf{IR} backend $\to$ JVM bytecode, JS, Wasm, or LLVM (Native).
  \item \textbf{Kotlin/Native}: LLVM machine code, a \textbf{tracing GC} (new memory
        manager, default since 1.7.20, no \texttt{freeze}); ARC only at the ObjC/Swift edge.
\end{itemize}

\section{Traps}
\begin{itemize}\raggedright
  \trap{``\texttt{List} is immutable'' — only a read-only \emph{interface}.}
  \trap{\texttt{foo\$default} in a trace = the default-args bridge.}
  \trap{Changed a library \texttt{const val}: callers keep the \emph{old} inlined value.}
\end{itemize}

\textbf{Remember:} \emph{FileKt · INSTANCE · Outer\$Companion · \$default + mask · indy
lambdas · List = java.util.List.}
