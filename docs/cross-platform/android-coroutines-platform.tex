\sheettitle{Kotlin coroutines \& Flow · Android vs iOS platform}{android · folio}

\oneliner{A coroutine is a \textbf{suspendable computation} launched in a \textbf{scope}
whose \texttt{Job} tree gives \textbf{structured concurrency} (a parent waits for and
cancels its children); \textbf{cancellation is cooperative}. \texttt{Flow} is a
\textbf{cold} stream, \texttt{StateFlow}/\texttt{SharedFlow} are \textbf{hot} — the Kotlin
twins of \texttt{async}/\texttt{await}, \texttt{Task}, \texttt{AsyncSequence} and Combine
subjects.}

\section{Coroutines — the mechanism}
\begin{itemize}\raggedright
  \item \texttt{suspend fun} compiles to a \textbf{state machine} + a hidden
        \texttt{Continuation} param; suspending frees the thread (like \texttt{await}).
  \item \texttt{launch} $\to$ \texttt{Job} (fire-and-forget) · \texttt{async} $\to$
        \texttt{Deferred<T>}, \texttt{await()}; \texttt{coroutineScope\{\}} waits for all
        children, rethrows the first failure; \texttt{supervisorScope\{\}} isolates them.
  \item \textbf{Scopes}: \texttt{viewModelScope} (\texttt{onCleared}),
        \texttt{lifecycleScope} (\texttt{onDestroy}), \texttt{rememberCoroutineScope}
        (leaves composition). \texttt{GlobalScope} = unstructured, leaks.
  \item \textbf{Dispatchers}: \texttt{Main} (UI; \texttt{Main.immediate} skips the re-post
        if already there) · \texttt{IO} (blocking, 64 threads or \#cores if more) ·
        \texttt{Default} (CPU, \#cores). \texttt{withContext(IO) \{\}} hops and returns.
  \item \textbf{Cancellation is cooperative}: \texttt{CancellationException} at the next
        suspension point; CPU loops call \texttt{ensureActive()}/\texttt{yield()}.
        Blocking I/O is not interrupted.
  \item Errors: \texttt{launch} propagates up the Job tree to a
        \texttt{CoroutineExceptionHandler} on the \textbf{root} (ignored on children);
        \texttt{async} rethrows at \texttt{await()}.
\end{itemize}

\section{Flow}
\begin{itemize}\raggedright
  \item \textbf{Cold} \texttt{flow\{\}}: producer re-runs \emph{per collector}; Room/Retrofit
        streams. \texttt{flowOn(IO)} moves \emph{upstream} only; \texttt{.catch} sees
        \emph{upstream} errors only.
  \item \textbf{StateFlow}: always a value, conflated, replays latest — screen state.
        \textbf{SharedFlow}: \texttt{replay}/buffer configurable — broadcasts.
  \item \texttt{stateIn(scope, WhileSubscribed(5\_000), init)}: upstream survives the
        rotation gap, stops 5 s after the last collector. \texttt{Eagerly}/\texttt{Lazily}
        never stop.
  \item \texttt{flatMapLatest} cancels the previous inner flow (search);
        \texttt{combine} = latest of each (state), \texttt{zip} pairs by index.
\end{itemize}

\section{Example — VM + safe collection}
\begin{lstlisting}[language=KotlinSheet]
val ui: StateFlow<Ui> = repo.devices()        // cold (Room)
  .map { Ui.Loaded(it) }
  .catch { emit(Ui.Error(it)) }              // upstream only
  .stateIn(viewModelScope, WhileSubscribed(5_000), Ui.Loading)
private val _ev = Channel<Event>(Channel.BUFFERED)
val events = _ev.receiveAsFlow()             // one-shot
fun refresh() = viewModelScope.launch {
  try { withContext(Dispatchers.IO) { repo.sync() } }
  catch (e: IOException) { _ev.send(Event.Offline) }
}                                   // never catch(Exception)
// Activity / Fragment (viewLifecycleOwner):
lifecycleScope.launch { repeatOnLifecycle(STARTED) {
  vm.ui.collect { render(it) } } }
\end{lstlisting}

\section{Traps}
\begin{itemize}\raggedright
  \trap{\texttt{catch (e: Exception)} swallows \texttt{CancellationException} — rethrow it or
        catch specific types.}
  \trap{\texttt{StateFlow} for one-shot events re-fires after rotation; a
        \texttt{SharedFlow(replay=0)} drops events with no collector $\to$ \texttt{Channel}.}
  \trap{Swift \texttt{Task\{\}} is \emph{not} a child: nobody cancels it — store and cancel.}
\end{itemize}

\section{Remember}

collect from STARTED.}
