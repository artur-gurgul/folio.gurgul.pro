\sheettitle{Progressive delivery — rolling, canary, shadow, rings, staged rollouts}{engineering · folio}

\oneliner{\textbf{Expose a new version to users in steps}, measure each step \textbf{against a baseline}, and
\textbf{promote or abort automatically} — so a bad release hurts 1–5\,\% of traffic for minutes instead of
everyone for hours. \textbf{Deploy $\ne$ release}: the binary can be everywhere while the feature is off.}

\hd{Rolling update — the default}
\begin{itemize}
  \item K8s \ct{RollingUpdate}: \ct{maxSurge} (extra pods, \% rounds \textbf{up}) and \ct{maxUnavailable} (\%
        rounds \textbf{down}), both default \textbf{25\,\%}, not both 0. 10 replicas: $\le$\,13 pods, $\ge$\,8
        ready. \ct{maxUnavailable: 0} = never below capacity.
  \item Only \textbf{readiness} gates progress (+ \ct{minReadySeconds}); nothing looks at error rates.
        \ct{progressDeadlineSeconds} (600) only marks the rollout \emph{failed} — \textbf{no automatic rollback}:
        \ct{kubectl rollout undo}.
  \item v1 and v2 serve \emph{at the same time} $\Rightarrow$ API, schema and messages compatible both ways.
  \item \textbf{Recreate} only when versions cannot coexist (singleton, exclusive lock).
\end{itemize}

\hd{Canary + automated analysis}
\begin{itemize}
  \item Route a \textbf{small weight} to v2 (mesh, ingress, Gateway API, ALB weights). No router
        $\Rightarrow$ weight $\approx$ replica ratio (1 of 10 pods = 10\,\%).
  \item Compare canary with a \textbf{baseline} of the \emph{old} version deployed \emph{at the same time, same
        size} — not with the big, warm production fleet (caches, JIT, heap differ).
  \item \textbf{Golden signals}: latency (p99, successes and errors separately), traffic, errors, saturation;
        plus restarts/OOM and a business metric (checkout success).
  \item \textbf{Kayenta} (Spinnaker): Mann–Whitney U per metric at 98\,\% confidence $\to$ Pass/High/Low;
        score = \% of metrics passing vs \emph{marginal} / \emph{pass} thresholds; critical metric fails
        $\to$ 0. Needs $\ge$\,50 data points per metric: steps last minutes–hours.
  \item \textbf{SLO burn rate}: error budget used per unit time; 14.4$\times$ over 1\,h (and 5\,min) = 2\,\% of a
        30-day budget $\Rightarrow$ abort.
\end{itemize}

\hd{Abort criteria — write them before the rollout}
Error rate or p99 worse than baseline beyond a set margin · burn rate over threshold · crash loops /
OOMKilled · saturation (CPU, pool, queue depth) · business KPI drop · \textbf{no data = fail}, not pass.
Abort = weight to 0, canary scaled down; investigate, don't retry.

\hd{Sticky routing}
Per-request \% sends one user to v1 then v2: flickering UI, lost session, v2-written data read by v1.
Route by \textbf{user/device id hash}, cookie or header (Flagger A/B mode; ALB target-group stickiness);
send \emph{internal} users via a header. Mobile: route by \ct{App-Version}.

\hd{Example — Argo Rollouts canary}
\begin{lstlisting}[language=YAMLSheet]
apiVersion: argoproj.io/v1alpha1
kind: Rollout
spec:
  replicas: 10
  strategy:
    canary:
      canaryService: api-canary     # router splits by weight
      stableService: api-stable
      trafficRouting: { nginx: { stableIngress: api } }
      analysis:                     # background: whole rollout
        templates: [{ templateName: success-rate }]
      steps:
      - setWeight: 5
      - pause: { duration: 15m }
      - setWeight: 25
      - pause: {}                   # waits for a human: promote
# AnalysisTemplate: successCondition: result[0] >= 0.99,
#   interval: 5m, failureLimit: 2  -> abort: weight 0, Degraded
\end{lstlisting}

\hd{Shadow, rings, flags}
\begin{itemize}
  \item \textbf{Shadow / dark launch}: mirror live requests to v2, compare responses offline. Great for rewrites
        and load; dangerous with side effects (double charge, double email) — stub writes.
  \item \textbf{Rings}: staff $\to$ beta $\to$ one region $\to$ all, each baked.
  \item \textbf{Flags}: deploy moves \emph{code}, a flag releases \emph{behaviour} — per user, reversible
        in seconds. Canary the binary, flag the feature.
\end{itemize}

\hd{Traps}
\begin{itemize}
  \trap{``K8s rolls back a bad rollout'' — it stops; undo is on you.}
  \trap{Canary vs the whole prod fleet instead of a same-age baseline.}
  \trap{5\,\% of a quiet hour = too few requests to judge anything.}
  \trap{Shadowing a write path: every payment happens twice.}
\end{itemize}

\hd{Remember}
\textbf{Small step, same-age baseline, pre-written abort.}
