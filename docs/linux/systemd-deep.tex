\sheettitle{systemd — units, dependencies, services, sandbox, journal}{linux · folio}

\oneliner{systemd (PID~1) turns \textbf{unit files} into a \textbf{transaction of jobs} on two
independent axes — \textbf{requirement} (\ct{Wants}, \ct{Requires}: \emph{what} starts) and
\textbf{ordering} (\ct{After}, \ct{Before}: \emph{when}) — and gives each service its own
\textbf{cgroup} to limit, log and kill \emph{all} its processes.}

\hd{Units, and where they come from}
\ct{.service} · \ct{.socket} · \ct{.timer} (both start the same-named service) · \ct{.path} (inotify)
· \ct{.target} (sync point) · \ct{.mount} named after its path (\ct{/srv/data} $\to$
\ct{srv-data.mount}) · \ct{.automount} · \ct{.swap} · \ct{.device} · \ct{.slice} (cgroup inner node)
· \ct{.scope} (processes systemd did not fork). First match wins: \ct{/etc/systemd/system.control}
(\ct{set-property}) $\to$ \ct{/run/systemd/transient} (\ct{systemd-run}) $\to$
\textbf{\ct{/etc/systemd/system}} (admin) $\to$ \ct{/run/systemd/system} $\to$ \ct{/usr/local/lib/…}
$\to$ \textbf{\ct{/usr/lib/systemd/system}} (package: never edit); \textbf{generators} (\ct{fstab}
$\to$ \ct{.mount}) fill \ct{/run/systemd/generator*} at boot and on every \ct{daemon-reload}.
\textbf{Drop-ins} \ct{app.service.d/*.conf} merge alphanumerically after the unit;
\ct{foo-bar.service} also reads \ct{foo-.service.d/}. Linked to \ct{/dev/null}: \textbf{masked},
never activated, not even as a dependency. \ct{\#} or \ct{;} comments only at the \emph{start} of a
line (\ct{KEY=v \#x} keeps ``\ct{\#x}''); an empty assignment resets a list.

\hd{Requirement directives — in \ct{[Unit]} of X}
{\scriptsize\setlength\tabcolsep{2pt}
\begin{tabular}{@{}>{\ttfamily}L{12mm}L{66mm}@{}}
\toprule
Wants= & start them too; their failure is ignored — ``the recommended way'' \\
Requires= & + their failed start fails X \emph{only if} X is \ct{After=} them; their explicit stop / restart stops / restarts X \\
Requisite= & must already be active, else X fails at once \\
BindsTo= & + X stops when they stop for \emph{any} reason; with \ct{After=}, X is active only while they are \\
PartOf= & their stop / restart propagates to X — one way, nothing else \\
Upholds= & (249) while X is up, restart them whenever found inactive or failed \\
Conflicts= & starting one stops the other \\
OnFailure= & started when X enters \textbf{failed} · \ct{OnSuccess=} (249) \\
\bottomrule
\end{tabular}}
\begin{itemize}
  \item \textbf{Ordering is orthogonal}: no \ct{After=}/\ct{Before=} $\Rightarrow$ \emph{parallel};
        stops run in inverse order, before starts. Services get \ct{Requires}+\ct{After}
        \ct{sysinit.target}, \ct{After} \ct{basic.target}, \ct{Conflicts}+\ct{Before}
        \ct{shutdown.target} unless \ct{DefaultDependencies=no}.
  \item \ct{network.target} is ``only very weakly defined'': \ct{Wants=} +
        \ct{After=network-online.target}. A \textbf{cycle} is broken by deleting a job, often an
        innocent one's. \ct{Condition*=} false skips; \ct{Assert*=} fails.
\end{itemize}

\hd{\ct{Type=} — when has the start finished?}
{\scriptsize\setlength\tabcolsep{2pt}
\begin{tabular}{@{}>{\ttfamily}L{12mm}L{66mm}@{}}
\toprule
simple & \ct{fork()} returned — default; reports success even if binary or \ct{User=} is missing \\
exec & \ct{execve()} succeeded — the man page's pick for long-running services \\
forking & the parent exited (legacy daemons; set \ct{PIDFile=}) \\
oneshot & the process \emph{exited}; default without \ct{ExecStart=}; may have several \\
dbus & \ct{BusName=} acquired \\
notify & \ct{READY=1} arrived; \ct{notify-reload} (253): reload = SIGHUP $\to$ \ct{RELOADING=1}, \ct{READY=1} \\
idle & after the other jobs, $\le$5\,s — console output only \\
\bottomrule
\end{tabular}}\par
Clients need the service's own IPC up? \ct{notify}, \ct{notify-reload} or \ct{dbus}.

\textbf{Clean} = exit 0, or SIGHUP, SIGINT, SIGTERM, SIGPIPE (not for oneshot). Never after
\ct{systemctl stop}.
\ct{RestartSec=} 100\,ms; \ct{RestartSteps=} + \ct{RestartMaxDelaySec=} (254) back off
exponentially; \ct{RestartRandomizedDelaySec=} (262) adds jitter.

\hd{Timeouts, stopping, \ct{Exec*=} lines}
\ct{TimeoutStartSec} / \ct{TimeoutStopSec} 90\,s (oneshot: no start timeout;
\ct{EXTEND\_TIMEOUT\_USEC=} buys time). \ct{WatchdogSec=}: no \ct{WATCHDOG=1} in time $\to$
\textbf{SIGABRT}. Stop: \ct{ExecStop=} $\to$ SIGTERM $\to$ \ct{TimeoutStopSec} $\to$ SIGKILL.
\ct{KillMode=}\textbf{\ct{control-group}} (default) signals every process in the cgroup — a
\ct{nohup cmd \&} child dies too · \ct{mixed}: TERM main, KILL the rest · \ct{process}, \ct{none}:
discouraged, processes escape. \ct{ExecStopPost=} runs even after a crash, given
\ct{\$SERVICE\_RESULT} (\ct{exit-code}, \ct{signal}, \ct{timeout}, \ct{oom-kill} …).
No shell: prefixes \ct{-} failure ignored · \ct{@} sets \ct{argv[0]} · \ct{:} no \ct{\$VAR}
expansion · \ct{+} full privileges · \ct{!} lifts only \ct{User=}/\ct{Group=} · \ct{|} (258) via the
shell. \ct{ExecCondition=}: 1–254 skips, 255 fails.

\hd{systemctl, systemd-analyze}
\ct{enable} = the \ct{[Install]} symlinks (\ct{WantedBy=} $\to$ a \ct{.wants/} link) + reload, does
\emph{not} start (\ct{\dd now} does) · \ct{edit} writes \ct{app.service.d/override.conf} and reloads
(\ct{\dd full}: a copy); a new command there needs an empty \ct{ExecStart=} first — two get the unit
refused · \ct{revert} undoes \ct{edit}, \ct{set-property}, \ct{mask} · \ct{cat} ·
\ct{show -p NRestarts,Result} · \ct{daemon-reload} re-runs generators, rebuilds the tree.
\ct{systemd-analyze blame} (own start time, ``might be misleading'': waiting counts) ·
\ct{critical-chain} (\ct{@} active at, \ct{+} took) · \ct{verify} (unknown directives, missing
binaries) · \ct{security} · \ct{calendar}.

\hd{sd\_notify, templates, sockets}
\begin{itemize}
  \item To \ct{\$NOTIFY\_SOCKET}: \ct{READY=1} · \ct{RELOADING=1} · \ct{STOPPING=1} · \ct{STATUS=} ·
        \ct{WATCHDOG=1} · \ct{FDSTORE=1}: fds kept across restarts (\ct{FileDescriptorStoreMax=}, 0).
  \item \ct{app@.service} + \ct{start app@8080} $\Rightarrow$ \ct{\%i}=\ct{8080}. \ct{\%n} name ·
        \ct{\%p} prefix · \ct{\%H} host · \ct{\%t} \ct{/run} · \ct{\%S} \ct{/var/lib} · \ct{\%d}
        credentials. \textbf{\ct{\%u}, \ct{\%h} = the manager's user}: \ct{root} in a system unit,
        not \ct{User=}.
  \item Sockets: fds from \textbf{3} with \ct{LISTEN\_PID}, \ct{LISTEN\_FDS}, \ct{LISTEN\_FDNAMES};
        bound before the service runs: start on demand, in parallel, restart without refusing
        clients. \ct{Accept=no} (default): one service gets the listener — ``preferable'' for
        performance; \ct{yes}: an \ct{app@.service} per connection (max 64). $>$20 triggers in
        2\,s fail the socket.
\end{itemize}

\hd{Timers instead of cron}
\ct{OnCalendar=} \ct{DayOfWeek Y-M-D h:m:s}; \ct{daily} = \ct{*-*-* 00:00:00}, \ct{weekly} =
\ct{Mon *-*-* 00:00:00}; \ct{/} repeats, \ct{..} ranges, \ct{\textasciitilde} from month end, zone
suffix. Monotonic: \ct{OnBootSec}, \ct{OnUnitActiveSec} \dots\ \ct{AccuracySec=} 1\,min ·
\ct{RandomizedDelaySec=} · \ct{RandomizedOffsetSec=} (258, stable) · \ct{Persistent=true}: a missed
calendar run fires on next activation · still running when it elapses $\Rightarrow$ \textbf{no
second run}.

\hd{User manager, transient units, images}
\begin{itemize}
  \item \ct{user@1000.service} = \ct{systemctl \dd user} (\ct{\textasciitilde/.config/systemd/user},
        \ct{WantedBy=default.target}); it ends with the \textbf{last session} unless
        \ct{loginctl enable-linger}: then it starts at boot and survives logout.
  \item \ct{systemd-run -p MemoryMax=1G \dd wait cmd}: a transient service; \ct{\dd scope}: run in the
        foreground by \ct{systemd-run} itself. \ct{run0} is the same binary (256).
  \item \ct{systemd-sysext} (248): read-only overlay of \ct{/usr} + \ct{/opt} from images whose
        \ct{extension-release} matches \ct{os-release}; \ct{systemd-confext} (254): \ct{/etc}.
        \ct{soft-reboot} (254): userspace only — kernel, \ct{/run}, sysctls stay.
\end{itemize}

\hd{journald + journalctl}
Every service's stdout / stderr, as fields (\ct{MESSAGE}, \ct{PRIORITY}, \ct{\_PID},
\ct{\_SYSTEMD\_UNIT}). \ct{Storage=} \ct{volatile} (\ct{/run/log/journal}) · \ct{persistent}
(\ct{/var/log/journal}) · \ct{auto} (persistent if that exists) · \ct{none}; default
\textbf{\ct{persistent}} since 259, a build option. Disk: 10\,\% used, 15\,\% kept free, each
$\le$4\,G; 10\,000 messages / 30\,s per service. \ct{journalctl -u app -b -1 -p err \dd since -1h
-g 'time.?out' -o json} · \ct{-f} · \ct{-k} · \ct{\_PID=42} · \ct{\dd vacuum-size=1G} (archived
files only).
