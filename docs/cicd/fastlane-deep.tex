\sheettitle{fastlane — config, lanes, match internals, store auth}{cicd · folio}

\oneliner{fastlane is a \textbf{Ruby gem} (pin it with \textbf{Bundler}) whose \textbf{lanes} call
\textbf{actions}; \ct{gym}, \ct{scan}, \ct{pilot}, \ct{deliver}, \ct{snapshot}, \ct{match},
\ct{supply} wrap \ct{xcodebuild}, Apple's upload tools, the \textbf{App Store Connect API} and the
\textbf{Google Play Developer API}. On CI its job is to make \textbf{signing and upload
non-interactive}: \ct{match} decrypts one shared cert/profile store into a \textbf{throw-away
keychain}, and an \textbf{API key (.p8)} replaces an Apple ID that needs 2FA.}

\hd{Lanes}
\begin{itemize}
  \item \ct{platform :ios do} $\to$ \ct{fastlane ios beta}; \ct{desc} feeds \ct{fastlane lanes} and the
        README; a \ct{private\_lane} is hidden and callable only from another lane.
  \item A lane call is a method call: \ct{build(release: true)}; last expression = return value;
        \ct{next} leaves early and \ct{after\_each}/\ct{after\_all} still run.
  \item \ct{before\_all}/\ct{after\_all} once; \ct{before\_each}/\ct{after\_each} around \textbf{every}
        lane, called ones too; \ct{error} on any raise.
  \item \ct{lane\_context[SharedValues::…]} chains actions: \ct{upload\_to\_testflight} defaults
        \ct{ipa} to \ct{IPA\_OUTPUT\_PATH} and the key to \ct{APP\_STORE\_CONNECT\_API\_KEY}.
  \item \textbf{Plugins} are gems (\ct{fastlane add\_plugin x}) — third-party code on the signing
        machine: pin and review. \ct{fastlane action gym} lists every option and its env name.
\end{itemize}

\hd{Tool $\leftrightarrow$ action}
{\scriptsize \ct{gym}=\ct{build\_app} · \ct{scan}=\ct{run\_tests} · \ct{pilot}=\ct{upload\_to\_testflight} ·
\ct{deliver}=\ct{upload\_to\_app\_store} · \ct{snapshot}=\ct{capture\_ios\_screenshots} ·
\ct{precheck}=\ct{check\_app\_store\_metadata} · \ct{match}=\ct{sync\_code\_signing} ·
\ct{cert}=\ct{get\_certificates} · \ct{sigh}=\ct{get\_provisioning\_profile} ·
\ct{supply}=\ct{upload\_to\_play\_store}. \textbf{Deprecated}: \ct{xcversion} $\to$ \ct{xcodes};
\ct{automatic\_code\_signing} $\to$ \ct{update\_code\_signing\_settings}. \textbf{precheck} runs inside
\ct{deliver} before submit at level \ct{warn}: placeholder text, other platforms, unreachable URLs.}

\hd{Example — Fastfile}
\begin{lstlisting}[language=RubySheet]
before_all { setup_ci }          # no-op unless a CI env var is set
platform :ios do
  lane :beta do
    app_store_connect_api_key(key_id: ENV["ASC_KEY_ID"],
      issuer_id: ENV["ASC_ISSUER_ID"], key_content: ENV["ASC_P8_B64"],
      is_key_content_base64: true)     # -> lane_context
    match(type: "appstore", readonly: true)
    increment_build_number(build_number: latest_testflight_build_number + 1)
    build_app(scheme: "App", export_method: "app-store")
    upload_to_testflight(skip_waiting_for_build_processing: true)
  end
end
\end{lstlisting}
\ct{increment\_build\_number} runs \ct{agvtool new-version -all <n>}; with no \ct{n}, \ct{next-version}
— which needs Apple Generic versioning. An empty train returns \ct{initial\_build\_number} (1).

\hd{match — what it writes and installs}
\begin{itemize}
  \item \ct{certs/<type>/<cert-id>.cer + .p12} (\ct{distribution} serves \ct{appstore} and
        \ct{adhoc}) · \ct{profiles/appstore/AppStore\_<bundle>.mobileprovision}; portal name
        \ct{match AppStore <bundle>}.
  \item \ct{security import -T /usr/bin/codesign} into \ct{keychain\_name} (default \ct{login.keychain},
        on CI \ct{setup\_ci}'s), then \ct{set-key-partition-list -S apple-tool:,apple:,codesign:} —
        signing without a UI prompt.
  \item Profiles go to \ct{\textasciitilde/Library/Developer/Xcode/UserData/Provisioning Profiles}
        for Xcode $\geq$ 16 (\ct{\textasciitilde/Library/MobileDevice/…} before; right only since
        \textbf{2.230.0}) — select Xcode (\ct{xcodes}) \emph{before} match.
  \item Sets \ct{ENV["sigh\_<bundle>\_appstore"]} = profile UUID (+ \ct{\_profile-name},
        \ct{\_team-id} …), \ct{MATCH\_PROVISIONING\_PROFILE\_MAPPING} for gym, \ct{SIGH\_PROFILE\_TYPE}
        as gym's default \ct{export\_method}.
  \item \ct{readonly} fetches only · \ct{force\_for\_new\_devices} renews when the portal's device
        count changed · \ct{change\_password} · \ct{import} (existing \ct{.cer}/\ct{.p12}) · \ct{migrate}
        (storage $\to$ storage) · \ct{nuke distribution} revokes + deletes: App Store / TestFlight builds
        keep working, \textbf{Ad Hoc and Enterprise builds stop}.
\end{itemize}

\hd{Auth — API key vs Apple ID}
{\scriptsize\setlength{\tabcolsep}{2pt}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{12mm}>{\raggedright\arraybackslash}p{66mm}@{}}
\toprule
\textbf{ASC API key} & Key ID + Issuer ID + \ct{.p8} (downloadable once). \textbf{Team} key: an Admin
  makes it, role-scoped, all apps. \textbf{Individual} key: no Provisioning, Sales and Finance or
  notaryTool — match, sigh, cert need a team key. Apple rejects tokens $>$20 min for most requests
  (scoped GET-only tokens on a few resources: up to 6 months).
  Works for pilot, deliver, match, sigh, cert, precheck (not IAP); \textbf{not} \ct{pem} \\
\textbf{Apple ID} & \ct{FASTLANE\_USER} + password + 2FA; \ct{fastlane spaceauth} $\to$
  \ct{FASTLANE\_SESSION}, valid 1 day – 1 month and only in the region it was made — make it on the CI
  machine. An app-specific password covers \textbf{binary upload only}. It drives Apple's web sign-in:
  2.240.0 had to re-find the sign-in widget key after Apple dropped its endpoint \\
\bottomrule
\end{tabular}}

\hd{supply — Google Play}
\begin{itemize}
  \item Service-account JSON (\ct{json\_key} / \ct{\_data}). One run = one Play Developer API
        \textbf{edit}: insert $\to$ upload AAB $\to$ track $\to$ listing $\to$ commit (or
        \ct{validate\_only}). \textbf{A commit or a Play Console change invalidates every other open
        edit} — serialise jobs.
  \item \ct{track} \ct{production} (default), \ct{beta}, \ct{alpha}, \ct{internal}; \ct{release\_status}
        \ct{completed} (default), \ct{draft}, \ct{halted}, \ct{inProgress}; \ct{rollout: 0.1} $\to$
        \ct{inProgress} + \ct{userFraction}, \ct{rollout: 1} $\to$ \ct{completed}.
\end{itemize}

\hd{Traps}
\begin{itemize}
  \trap{\ct{sh("ls")} runs in \ct{./fastlane}, actions in the root — relative paths differ.}
  \trap{No \ct{bundle exec} = the global fastlane; plugins may not load.}
  \trap{A committed \ct{.env.ci} \emph{overrides} the CI secret of the same name.}
  \trap{\ct{-showBuildSettings} times out on big projects (3\,s × 3):
        \ct{FASTLANE\_XCODEBUILD\_SETTINGS\_TIMEOUT}.}
  \trap{\ct{export\_method} takes only \ct{app-store}, \ct{ad-hoc} \ldots; Xcode 15.4+ logs them as
        deprecated (\ct{app-store-connect}, \ct{release-testing}).}
