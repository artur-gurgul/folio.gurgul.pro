\sheettitle{Video playback — HLS, ABR, AVPlayer in production}{ios-platform · folio}

\oneliner{\textbf{HLS} = plain HTTP files: a \textbf{multivariant playlist} lists the same video
at several bitrates, each variant is a \textbf{media playlist} of short \textbf{segments}
(\ct{.ts} or fMP4). The player measures throughput and buffer and \textbf{switches variant at
segment boundaries} (ABR). In AVFoundation you watch \textbf{state} —
\ct{AVPlayerItem.status}, \ct{AVPlayer.timeControlStatus} + \ct{reasonForWaitingToPlay} — and
tune \textbf{startup, caps and buffer}, never the download itself.}

\hd{HLS (RFC 8216)}
\begin{itemize}
  \item \textbf{Multivariant} playlist: \ct{EXT-X-STREAM-INF} per variant — \ct{BANDWIDTH} (peak
        bit/s, required), \ct{AVERAGE-BANDWIDTH}, \ct{CODECS}, \ct{RESOLUTION}, \ct{FRAME-RATE}, group
        refs; \ct{EXT-X-MEDIA} = alternate audio/subtitle renditions (\ct{GROUP-ID}, \ct{LANGUAGE});
        \ct{EXT-X-I-FRAME-STREAM-INF} = key frames only, for scrubbing thumbnails.
  \item \textbf{Media} playlist: \ct{EXT-X-TARGETDURATION}, \ct{EXT-X-MEDIA-SEQUENCE}, \ct{EXTINF}
        per segment. \textbf{VOD} ends with \ct{EXT-X-ENDLIST}; \textbf{live} = sliding window the
        client re-fetches about every target duration; \ct{EVENT} = append-only.
        \ct{EXT-X-DISCONTINUITY} at an ad splice.
  \item \textbf{Segments}: MPEG-TS or fMP4/CMAF + init (\ct{EXT-X-MAP}). \textbf{Keys}:
        \ct{EXT-X-KEY} \ct{AES-128} / \ct{SAMPLE-AES}; FairPlay = \ct{skd://} URI.
  \item \textbf{LL-HLS} (8216\textbf{bis} draft): \textbf{partial segments} (\ct{EXT-X-PART}, sub-second), \ct{PRELOAD-HINT},
        \textbf{blocking reload} (\ct{?\_HLS\_msn=…\&\_HLS\_part=…} held until it exists; server says
        \ct{CAN-BLOCK-RELOAD=YES}), \ct{RENDITION-REPORT}, delta playlists (\ct{EXT-X-SKIP}).
        Latency $\approx$ TV broadcast.
\end{itemize}

\hd{ABR — what you control}
\begin{itemize}
  \item The player estimates bandwidth from segment downloads + buffer health and picks the start
        variant itself; \ct{startsOnFirstEligibleVariant} (iOS 14) forces the first listed one.
  \item Caps on \ct{AVPlayerItem}: \ct{preferredPeakBitRate} (bit/s, 0 = none) ·
        \ct{…ForExpensiveNetworks} (iOS 15, cellular) · \ct{preferredMaximumResolution}.
  \item \ct{preferredForwardBufferDuration} (s; 0 = system decides): low = fast start, more
        stalls; high = memory + bytes wasted when the user scrolls away.
\end{itemize}

\hd{AVFoundation state — watch, don't poll}
\begin{itemize}
  \item \ct{AVURLAsset} (media; \ct{try await asset.load(.isPlayable, .duration)}) $\to$
        \ct{AVPlayerItem} (presentation: \ct{status} \ct{.unknown/.readyToPlay/.failed},
        \ct{isPlaybackLikelyToKeepUp}, \ct{loadedTimeRanges}, logs) $\to$ \ct{AVPlayer} (rate,
        \ct{timeControlStatus} \ct{.paused/.waitingToPlayAtSpecifiedRate/.playing}) $\to$
        \ct{AVPlayerLayer} · \ct{AVPlayerViewController} · SwiftUI \ct{VideoPlayer}.
  \item \ct{reasonForWaitingToPlay}: \ct{.toMinimizeStalls}, \ct{.evaluatingBufferingRate},
        \ct{.noItemToPlay}, \ct{.interstitialEvent}, \ct{.waitingForCoordinatedPlayback}.
  \item \ct{automaticallyWaitsToMinimizeStalling} (default \textbf{true}): \ct{play()} may wait for
        buffer. \ct{false} = start at once; an empty buffer then \emph{pauses} (rate 0).
        Loops: \ct{AVPlayerLooper} + \ct{AVQueuePlayer}.
\end{itemize}

\hd{Example — observe, cap, measure}
\begin{lstlisting}[language=SwiftSheet]
let item = AVPlayerItem(asset: AVURLAsset(url: hlsURL))
item.preferredForwardBufferDuration = 2          // feed: start fast
item.preferredPeakBitRateForExpensiveNetworks = 1_500_000
let player = AVPlayer(playerItem: item)
obs = [item.observe(\.status) { it, _ in
         if it.status == .failed { report(it.error) } },
       player.observe(\.timeControlStatus) { p, _ in
         spin(p.timeControlStatus == .waitingToPlayAtSpecifiedRate) },
       layer.observe(\.isReadyForDisplay) { l, _ in    // hide poster
         poster.isHidden = l.isReadyForDisplay }]
NotificationCenter.default.addObserver(forName: .AVPlayerItemNewAccessLogEntry,
    object: item, queue: .main) { _ in
  guard let e = item.accessLog()?.events.last else { return }
  qoe.send(e.startupTime, e.numberOfStalls, e.indicatedBitrate) }
player.play()
\end{lstlisting}

\hd{Startup time and feeds}
\begin{itemize}
  \item \textbf{Start fast}: create the asset/item \emph{before} the tap (prefetch the next cells),
        load keys async, short buffer target, a low first variant, show a \textbf{poster} until
        \ct{isReadyForDisplay}. Measure time-to-first-frame, not \ct{play()} returning.
  \item \textbf{Feeds}: a \textbf{pool} of 2–3 \ct{AVPlayer}s, not one per cell; swap with
        \ct{replaceCurrentItem(with:)}; only the most-visible cell plays (muted); set off-screen items
        to \ct{nil} — each item holds buffers and a decoder. Cap resolution to the cell's pixels.
\end{itemize}

\hd{Background, PiP, AirPlay}
\begin{itemize}
  \item \textbf{Background audio}: \ct{AVAudioSession} \ct{.playback} + \ct{UIBackgroundModes: audio}. For a
        video, \ct{playerLayer.player = nil} on background (QA1668), re-attach on foreground.
  \item \textbf{PiP}: \ct{AVPictureInPictureController}, same mode;
        \ct{canStartPictureInPictureAutomaticallyFromInline} (14.2). \ct{AVPlayerViewController}: free.
  \item \textbf{AirPlay}: \ct{AVRoutePickerView}; with \ct{allowsExternalPlayback} the Apple TV
        fetches the stream itself — its URLs and auth must work from there.
\end{itemize}

\hd{DRM, offline, metrics}
\begin{itemize}
  \item \textbf{FairPlay}: \ct{AVContentKeySession(keySystem: .fairPlayStreaming)} — key request $\to$
        \ct{makeStreamingContentKeyRequestData} = \textbf{SPC} $\to$ your key server $\to$ \textbf{CKC}
        $\to$ \ct{processContentKeyResponse}. Offline needs a \emph{persistable} key.
  \item \textbf{Offline}: \ct{AVAssetDownloadURLSession} + \ct{makeAssetDownloadTask(downloadConfiguration:)};
        the system picks the location — store the URL it returns.
  \item \textbf{QoE}: \ct{accessLog()} events (\ct{indicatedBitrate}, \ct{observedBitrate},
        \ct{numberOfStalls}, \ct{startupTime}, \ct{numberOfDroppedVideoFrames}) + \ct{errorLog()}; not
        KVO — use the notifications. KPIs: startup time, rebuffer ratio, bitrate, fatal errors.
\end{itemize}

\hd{Traps}
\begin{itemize}
  \trap{Waiting on \ct{AVPlayer.status} — it's the \emph{item's} \ct{status} that means ready.}
  \trap{Spinner from \ct{rate == 0} — rate is the \emph{requested} speed; use \ct{timeControlStatus}.}
  \trap{New \ct{AVPlayer} per cell: memory, decoders, hitches. One item in two players: exception.}
  \trap{KVO token not stored $\to$ no callbacks; time observer never removed $\to$ leak.}
  \trap{\ct{isPlaybackLikelyToKeepUp} can be false with a \emph{full} buffer — it's a prediction.}
\end{itemize}

\hd{Remember}
\textbf{Playlist of playlists of segments} · the player picks, you \emph{cap} · watch the
\emph{item} for ready, the \emph{player} for waiting · pool players in feeds.
