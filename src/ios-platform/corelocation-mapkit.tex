\sheettitle{Core Location + MapKit}{ios-platform · folio}

\oneliner{Core Location fuses GPS/Wi-Fi/cell into fixes under a permission the user can
\textbf{shrink any time} (While Using/Always \emph{and} Precise/Approximate); the right
service is the \textbf{cheapest one that answers the question}. MapKit draws
it: \textbf{annotations} = points (reused views), \textbf{overlays} = shapes (renderers).}

\section{Core Location}
\begin{itemize}
  \item \texttt{CLLocationManager}: strong ref, created on main (callbacks use the
        creating thread's run loop). \texttt{locationManagerDidChangeAuthorization}
        (iOS 14) fires at creation too — read \texttt{authorizationStatus} there.
  \item Info.plist \texttt{NSLocationWhenInUseUsageDescription} (+
        \texttt{\dots AlwaysAndWhenInUse\dots} for Always); missing $\to$ no prompt.
  \item \texttt{requestLocation()} = one fix, then stops (needs
        \texttt{didFailWithError}). \texttt{startUpdatingLocation()} = stream, tuned
        by \texttt{desiredAccuracy}, \texttt{distanceFilter}, \texttt{activityType}.
  \item \textbf{iOS 17}: \texttt{CLLocationUpdate.liveUpdates()} =
        \texttt{AsyncSequence}, no delegate; a held
        \texttt{CLBackground\allowbreak ActivitySession} keeps it alive in the background. \textbf{\texttt{CLMonitor}} (actor) replaces
        \texttt{startMonitoring(for:)}: \texttt{add(condition, identifier:)}, then
        iterate \texttt{events} (\texttt{.satisfied}/\texttt{.unsatisfied}).
  \item \textbf{iOS 18}: \texttt{CLServiceSession(authorization: .whenInUse)} declares what
        a feature \emph{needs}; Core Location prompts as appropriate. Iterating
        \texttt{liveUpdates}/\texttt{events} holds one implicitly (off with
        \texttt{NSLocationRequireExplicitServiceSession}). \textbf{Always} only takes
        effect while a \texttt{.always} session is held — started in the foreground.
  \item \textbf{Background}: mode \texttt{location} + \texttt{allowsBackground\allowbreak
        LocationUpdates = true}, started in the foreground.
  \item Quality: \texttt{horizontalAccuracy < 0} = invalid; first fix often cached
        (check \texttt{timestamp}). \texttt{CLGeocoder}: network, rate-limited.
\end{itemize}

\section{MapKit}
\begin{itemize}
  \item \textbf{Annotation} (\texttt{MKAnnotation} point) $\to$ \texttt{viewFor:}
        $\to$ \textbf{reused} \texttt{MKAnnotationView} (\texttt{register} +
        \texttt{dequeueReusable\dots}; \texttt{MKMarkerAnnotationView}).
        \textbf{Overlay} (\texttt{MKPolyline}/\allowbreak\texttt{Polygon}/\allowbreak
        \texttt{Circle}/\allowbreak\texttt{MKTileOverlay}) $\to$ \texttt{rendererFor:}
        $\to$ \texttt{MKOverlayRenderer}, scales with zoom.
  \item \textbf{Clustering} (iOS 11): shared \texttt{clusteringIdentifier} $\to$
        \texttt{MKClusterAnnotation}; \texttt{displayPriority}.
  \item \textbf{Search}: \texttt{MKLocalSearchCompleter} (as you type) $\to$
        \texttt{MKLocalSearch} $\to$ \texttt{[MKMapItem]}. \textbf{Directions}:
        \texttt{MKDirections.Request} $\to$ \texttt{calculate} $\to$ \texttt{MKRoute}
        (\texttt{polyline}, \texttt{expectedTravelTime}); or hand off:
        \texttt{openInMaps(\allowbreak launchOptions:)}.
  \item \textbf{SwiftUI \texttt{Map} (iOS 17)}: content builder (\texttt{Marker},
        \texttt{Annotation}, \texttt{MapPolyline}, \texttt{MapCircle},
        \texttt{UserAnnotation}), \texttt{MapCameraPosition} (\texttt{.automatic},
        \texttt{.region}, \texttt{.userLocation(\allowbreak fallback:)}),
        \texttt{.onMapCameraChange}, \texttt{.mapControls}. iOS 16
        \texttt{Map(coordinateRegion:)} is deprecated.
\end{itemize}

\section{Example — iOS 17 async + SwiftUI Map}
\begin{lstlisting}[language=SwiftSheet]
let bg = CLBackgroundActivitySession() // hold: bg updates
for try await u in CLLocationUpdate.liveUpdates() {
  guard let loc = u.location else { continue }
  track(loc) }                         // + u.isStationary
let m = await CLMonitor("stores")      // actor, 20 max
await m.add(CLMonitor.CircularGeographicCondition(
  center: shop, radius: 150), identifier: "shop-7")
for try await e in await m.events {
  if case .satisfied = e.state { arrived(e.identifier) } }
Map(position: $camera, selection: $picked) {
  ForEach(shops) { s in
    Marker(s.name, coordinate: s.coord).tag(s.id) }
  MapPolyline(route.polyline).stroke(.blue, lineWidth: 4)
  UserAnnotation() }
\end{lstlisting}

\section{Traps}
\begin{itemize}
  \trap{Reading \texttt{authorizationStatus} right after \texttt{request\dots} —
        still \texttt{.notDetermined}; the answer comes in the delegate.}
  \trap{No first prompt offers ``Always''. A cold \texttt{requestAlways} gives
        \emph{provisional} Always; the upgrade prompt shows \textbf{once} — after
        that, and after Don't Allow, only Settings.}
  \trap{\textbf{Blue pill}: mandatory for While-Using background use;
        \emph{off} by default for Always (opt in via
        \texttt{showsBackground\allowbreak LocationIndicator}).}
  \trap{\texttt{allowsBackgroundLocationUpdates} without the \texttt{location}
        mode crashes; \texttt{pausesLocationUpdatesAutomatically} silently stops a
        background tracker.}
  \trap{\textbf{20 regions} per app: monitor the nearest 19 + one big region round
        the user whose \emph{exit} re-plans. Geofences lag by minutes; radii under
        $\approx$100 m are unreliable.}
  \trap{Returning a view for \texttt{MKUserLocation} hides the blue dot — return
        \texttt{nil}. \texttt{CLGeocoder} per keystroke is throttled.}
\end{itemize}
