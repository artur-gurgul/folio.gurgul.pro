\sheettitle{Android lifecycle, ViewModel \& Jetpack Compose}{android · folio}

\oneliner{An Activity is \textbf{destroyed and recreated} on every configuration change
(rotation, locale, dark mode, window size); the \textbf{ViewModel} survives that, the
\textbf{saved-state Bundle} survives \textbf{process death}, and Compose redraws UI as a
\textbf{function of state} — re-running only the composables that \emph{read} a changed
\texttt{State}.}

\section{Lifecycle, ViewModel, process death}
\begin{itemize}\raggedright
  \item Config change $\Rightarrow$ \texttt{onPause/onStop/onDestroy} + a \textbf{new}
        Activity \texttt{onCreate(savedInstanceState)}: fields and views gone.
        \texttt{android:configChanges} opts out — avoid.
  \item \textbf{Fragment}: its \emph{view} dies (\texttt{onDestroyView}) while the instance
        sits on the back stack $\Rightarrow$ observe with \texttt{viewLifecycleOwner}.
  \item \textbf{ViewModel} lives in a \texttt{ViewModelStore} (Activity, Fragment or nav
        entry) carried across recreation; \texttt{onCleared()} only when the owner
        \emph{finishes}. \texttt{viewModelScope} = \texttt{SupervisorJob} +
        \texttt{Main.immediate}, cancelled there.
  \item \textbf{Process death}: killed in background; on return a new process rebuilds the
        Activity \emph{from the Bundle} + a new VM. \texttt{SavedStateHandle} (VM) and
        \texttt{rememberSaveable} (UI) write that Bundle — ids, not data (Binder
        $\approx$1 MB $\to$ \texttt{TransactionTooLargeException}).
  \item Activity \texttt{Context} in a VM leaks it per rotation: \texttt{applicationContext}.
\end{itemize}

\section{Compose — the model}
\begin{itemize}\raggedright
  \item Phases: \textbf{composition $\to$ layout $\to$ draw}. Snapshot state records which
        scope \textbf{read} which \texttt{State}; a write re-runs only those scopes.
  \item \texttt{remember} survives recomposition only; \texttt{remember(key)} recomputes on
        key change; \texttt{rememberSaveable} also survives rotation + death (Bundle types
        or a \texttt{Saver}). \textbf{Hoisting}: stateless \texttt{Foo(value, onValueChange)}.
  \item \textbf{Skipping}: skipped if all params are \textbf{stable} and equal. Stable:
        primitives, \texttt{String}, lambdas, \texttt{val}-only classes,
        \texttt{@Immutable}/\texttt{@Stable}. Unstable: \texttt{var}, \texttt{List}
        (interface), types from non-Compose modules. \textbf{Strong skipping} (default,
        Kotlin 2.0.20+): unstable params compared by identity, lambdas auto-remembered.
  \item \textbf{Modifier order} wraps: \texttt{size(100.dp).padding(10.dp)} = 80 dp content;
        reversed = 120 dp total. \texttt{clickable} first = padding tappable.
\end{itemize}

\section{Example — VM + hoisted screen}
\begin{lstlisting}[language=KotlinSheet]
class SearchVM(saved: SavedStateHandle, val repo: Repo)
    : ViewModel() {
  val query = saved.getStateFlow("q", "") // survives death
  private val _ui = MutableStateFlow<Ui>(Ui.Idle)
  val ui = _ui.asStateFlow()              // read-only
  fun onQuery(q: String) { saved["q"] = q
    viewModelScope.launch { _ui.value = repo.search(q) } }
}
@Composable fun Search(vm: SearchVM = viewModel()) {
  val q by vm.query.collectAsStateWithLifecycle()
  SearchBar(q, onChange = vm::onQuery)  // hoisted
  LaunchedEffect(q) { delay(300); log(q) } // per q
}
\end{lstlisting}

\section{Traps}
\begin{itemize}\raggedright
  \trap{\textbf{Rotation recreates the Activity} — the iOS reflex ``the VC stays'' loses
        state. Three tiers: \texttt{remember} < ViewModel < Bundle.}
  \trap{VM does \textbf{not} survive process death. ``Don't keep activities'' only destroys
        Activities; real death: \texttt{adb shell am kill <pkg>} while backgrounded.}
  \trap{\texttt{mutableStateOf(0)} without \texttt{remember} resets each recomposition.
        \texttt{LaunchedEffect(Unit)} never restarts (stale id).}
  \trap{Fast state (\texttt{scrollState.value}) read high up recomposes the screen — defer
        the read: lambda, \texttt{graphicsLayer\{\}}, \texttt{derivedStateOf}.}
  \trap{Bodies re-run often, in any order: no I/O or writes.}
\end{itemize}

\section{Remember}
\emph{remember < ViewModel < Bundle · state down, events up · read late, key your effects.}
