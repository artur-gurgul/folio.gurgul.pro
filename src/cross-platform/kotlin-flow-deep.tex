\sheettitle{Kotlin Flow — contract, operators, backpressure, hot flows}{cross-platform · folio}

\oneliner{A \texttt{Flow} is a \textbf{cold, sequential} stream: \texttt{collect} runs the
producer \emph{in the collector's coroutine}, each value passes every operator before the next
is produced, and the producer suspends while the collector works (natural backpressure). Two
rules make operators composable: \textbf{context preservation} (only \texttt{flowOn} changes
where upstream runs) and \textbf{exception transparency} (never emit from inside a
\texttt{catch}). Concurrency and hotness are \emph{opt-in}: \texttt{buffer}, \texttt{flowOn},
channels, \texttt{stateIn}/\texttt{shareIn}.}

\section{The contract}
\begin{itemize}\raggedright
  \item \textbf{Intermediate} ops (\texttt{map}, \texttt{filter}, \texttt{transform},
        \texttt{onEach}, \texttt{debounce}) return a new \texttt{Flow}, run nothing.
        \textbf{Terminal} ops (\texttt{collect}, \texttt{first}, \texttt{single},
        \texttt{toList}, \texttt{reduce}) are \texttt{suspend} and \emph{start} it;
        \texttt{launchIn(scope)} = \texttt{scope.launch \{ collect() \}}.
  \item \textbf{Context preservation}: \texttt{emit} only from the collecting coroutine —
        \texttt{withContext}/\texttt{launch} inside \texttt{flow \{\}} throws
        \texttt{IllegalStateException} (``Flow invariant is violated''). Other threads/
        coroutines $\to$ \texttt{channelFlow}/\texttt{callbackFlow}.
  \item \textbf{Exception transparency}: if \texttt{emit} throws (downstream failed), stop
        and rethrow; never swallow it with \texttt{try/catch} around \texttt{emit}. Use
        \texttt{catch \{\}} — it sees \textbf{upstream} only and may \texttt{emit} a fallback.
  \item Don't implement \texttt{Flow} yourself (not stable for inheritance) — use
        \texttt{flow \{\}} or \texttt{AbstractFlow}.
\end{itemize}

\section{Combining}
{\scriptsize
\noindent\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{19mm}X@{}}
\toprule
\texttt{combine(a,b)} & latest of each; first output once \emph{all} have emitted; then on any
  emission — UI state from several sources \\
\texttt{zip(a,b)} & pairs by index (1st with 1st); ends when either ends \\
\texttt{merge(a,b)} & interleaves values as they come, no pairing \\
\texttt{flatMapConcat} & inner flows one after another (order kept) \\
\texttt{flatMapMerge} & inner flows concurrently, \texttt{concurrency} = 16 by default \\
\texttt{flatMapLatest} & new outer value \textbf{cancels} the running inner flow — search box \\
\bottomrule
\end{tabularx}}

\section{Errors \& completion}
\begin{itemize}\raggedright
  \item \texttt{retry(3) \{ it is IOException \}} / \texttt{retryWhen \{ cause, attempt \ar\ … \}}
        \emph{re-collect upstream} (cold $\Rightarrow$ re-runs the request).
  \item \texttt{onCompletion \{ cause \ar\ \}}: runs on success, failure or cancel, sees
        upstream \emph{and} downstream errors, does \emph{not} handle them.
  \item Collector exceptions: move the work into \texttt{onEach}, then \texttt{catch},
        then \texttt{launchIn}.
\end{itemize}

\section{Example — callback, search, screen state}
\begin{lstlisting}[language=KotlinSheet]
fun LocationClient.updates(): Flow<Loc> = callbackFlow {
  val cb = LocationCallback { loc -> trySend(loc) } // non-suspending
  requestUpdates(cb)
  awaitClose { removeUpdates(cb) }  // REQUIRED: else ISE on close
}.conflate()                        // keep only the newest fix
val results = query                 // MutableStateFlow<String>
  .debounce(300).distinctUntilChanged()
  .flatMapLatest { q -> repo.search(q) }   // old search cancelled
  .catch { emit(emptyList()) }
val ui = combine(results, prefs) { r, p -> Ui(r, p) }
  .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), Ui())
\end{lstlisting}

\section{Hot flows — the knobs}
{\scriptsize
\noindent\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{11mm}>{\raggedright\arraybackslash}p{26mm}L@{}}
\toprule
 & \textbf{StateFlow} & \textbf{SharedFlow} (\texttt{replay}, \texttt{extraBufferCapacity},
  \texttt{onBufferOverflow} = 0, 0, \texttt{SUSPEND}) \\
\midrule
read & \texttt{.value}; \texttt{update\{\}} = CAS & \texttt{replayCache} \\
equal value & \textbf{not} emitted & emitted \\
slow sub & jumps to newest & buffer, then \texttt{SUSPEND}/\texttt{DROP\_OLDEST}/\texttt{DROP\_LATEST} \\

\bottomrule
\end{tabularx}}
{\raggedright StateFlow = SharedFlow(replay 1, \texttt{DROP\_OLDEST}) + \texttt{distinctUntilChanged}
+ initial value; neither ever completes.\par}

\begin{itemize}\raggedright
  \item \texttt{stateIn}/\texttt{shareIn}: one cold upstream $\to$ one shared collection.
        \texttt{Eagerly} (now, forever), \texttt{Lazily} (1st subscriber, forever),
        \texttt{WhileSubscribed(stopTimeoutMillis=0, replayExpirationMillis=MAX)}.
        \textbf{Why 5\,000}: a rotation gap is far shorter, so upstream survives it; in
        background it stops after 5 s.
  \item \textbf{Channels} (hot, each element to \emph{one} receiver): \texttt{Channel()} =
        \texttt{RENDEZVOUS} (0) · \texttt{BUFFERED} (64) · \texttt{CONFLATED} ·
        \texttt{UNLIMITED}. \texttt{callbackFlow}/\texttt{channelFlow} buffer 64;
        \texttt{channelFlow} may \texttt{launch} concurrent senders.
\end{itemize}

\section{Testing}
\begin{itemize}\raggedright
  \item \texttt{runTest}: virtual time (\texttt{delay}/\texttt{debounce} skipped; 60 s
        timeout); \texttt{advanceTimeBy}/\texttt{advanceUntilIdle}; endless collectors in
        \texttt{backgroundScope}; \texttt{Dispatchers.setMain(testDispatcher)}.
  \item \textbf{Turbine}: \texttt{flow.test \{ awaitItem(); awaitComplete() \}},
        \texttt{expectNoEvents()}. Assert a \texttt{StateFlow}'s \emph{values} — conflation
        may skip some.
\end{itemize}

\section{Traps}
\begin{itemize}\raggedright
  \trap{\texttt{MutableSharedFlow<Event>()} + \texttt{tryEmit}: with a subscriber it
        \textbf{fails} (no buffer) — set \texttt{extraBufferCapacity}, or \texttt{emit}.}
  \trap{\texttt{state.value = list.apply \{ add(x) \}} — same instance, \texttt{equals} true:
        nothing emitted. Publish a \textbf{new} value.}
  \trap{\texttt{fun items() = flow.stateIn(...)} — a new shared coroutine per call; use a \texttt{val}.}
  \trap{\texttt{catch} never sees the collector's own exceptions.}
\end{itemize}

\textbf{Remember:} \emph{cold + sequential · flowOn = upstream · buffer splits, conflate
drops, latest cancels.}
