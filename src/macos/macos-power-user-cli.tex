\sheettitle{macOS power-user CLI — what each command really talks to}{macos · folio}

\oneliner{Most macOS admin commands are \textbf{thin clients of a daemon} that owns and caches
the state — so change it \emph{through the tool}, name the \textbf{domain} (prefs domain,
launchd \ct{gui/<uid>}, log subsystem), and expect \textbf{SIP}, \textbf{TCC} and the
\textbf{sealed system volume} to refuse even \ct{root}.}

\hd{defaults — preferences through cfprefsd}
\begin{itemize}
  \item \textbf{Domain} = bundle id; \ct{-g} = \ct{NSGlobalDomain} (every app);
        \ct{-currentHost} = this host only. A file \emph{path} as domain: slated to go.
  \item No type flag $\Rightarrow$ \textbf{string}: \ct{-bool -int -float -date -data
        -array(-add) -dict(-add)}; check with \ct{read-type}. \ct{find <word>} ·
        \ct{export <dom> -} (XML) · \ct{delete-all} (every container).
  \item Since 10.8 cfprefsd caches and writes \emph{asynchronously}: a hand-edited plist ``will
        not necessarily be read, and may even be overwritten''.
\end{itemize}

\hd{launchctl — verbs on domain targets (\ct{U=\$(id -u)})}
{\scriptsize\setlength\tabcolsep{2.5pt}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{9mm}>{\raggedright\arraybackslash}p{56mm}>{\raggedright\arraybackslash}p{10mm}@{}}
\toprule
\textbf{want} & \textbf{modern} & \textbf{legacy}\\ \midrule
load & \ct{bootstrap gui/\$U \tl/Library/LaunchAgents/x.plist} & \ct{load}\\
unload & \ct{bootout gui/\$U/com.example.x} & \ct{unload}\\
restart & \ct{kickstart -k -p gui/\$U/com.example.x} (\ct{-p}: pid) & —\\
inspect & \ct{print gui/\$U/…} · \ct{blame} (why it was launched) & \ct{list}\\
keep off & \ct{disable gui/\$U/…} — persists across boots & \ct{-w}\\
\bottomrule
\end{tabular}}
\begin{itemize}
  \item \ct{system/} (anyone reads, root modifies) · \ct{user/<uid>/} (exists without a
        login) · \ct{gui/<uid>/} (the login session: agents) · \ct{login/<asid>/} ·
        \ct{pid/<pid>/} (the XPC services it can reach). Legacy verbs pick the domain by
        whether you are root.
  \item Relative \ct{ProgramArguments[0]} resolves via \ct{\_PATH\_STDPATH} =
        \ct{/usr/bin:/bin:/usr/sbin:/sbin}, not your \ct{PATH}; never \ct{daemon(3)} / fork + exit.
  \item Background items (13+): \ct{sfltool dumpbtm} lists, \ct{resetbtm} resets.
        \ct{brew services} = \ct{\tl/Library/LaunchAgents} (\ct{sudo}: \ct{/Library/LaunchDaemons}).
\end{itemize}

\hd{log — the unified log (10.12+)}
\begin{itemize}
  \item \ct{log stream \dd level debug} = live (debug \pdfonly{$\supset$}\htmlonly{⊃} info \pdfonly{$\supset$}\htmlonly{⊃} default) ·
        \ct{log show \dd last 1h} = stored, default level unless \ct{\dd info \dd debug} ·
        \ct{\dd last boot} · \ct{\dd style compact|json|ndjson}.
  \item Keys: \ct{subsystem}, \ct{category} (with subsystem), \ct{process}, \ct{sender}
        (the library), \ct{messageType} (default info debug error fault),
        \ct{eventMessage CONTAINS[c] "x"}. \ct{\tl/.logrc}: default options + predicate aliases.
  \item Persist debug for one subsystem: \ct{sudo log config \dd subsystem com.example.app
        \dd mode "level:debug,persist:debug"} (\ct{\dd status}, \ct{\dd reset}).
  \item \ct{sudo log collect \dd last 30m \dd output x.logarchive} (\ct{\dd device}: a paired
        device); store = \ct{/var/db/diagnostics}. 27.0 archives need 26.2+ to read.
  \item \ct{<private>} lifts only via \ct{OSLogPreferences} (Info.plist) or a
        \ct{com.apple.system.logging} profile (\ct{Enable-Private-Data}) — on a throwaway Mac.
\end{itemize}

\hd{Spotlight · open · clipboard}
\begin{itemize}
  \item \ct{mdfind 'kMDItemCFBundleIdentifier \eqq{} "com.example.app"'} — every copy (stale
        builds); \ct{-onlyin <dir>} \ct{-name} \ct{-live} \ct{-0}.
        \ct{mdls -name kMDItemWhereFroms f} = download URL. \ct{sudo mdutil -E /} rebuilds,
        \ct{-s} status.
  \item \ct{open -a Xcode .} · \ct{-b <id>} · \ct{-n} new instance · \ct{-W} wait · \ct{-R}
        reveal · \ct{-j} hidden · \ct{-F} no restore · \ct{\dd env K=V} · \ct{\dd args …}.
  \item \ct{pbpaste | jq . | pbcopy}; \ct{-pboard find}. Without a UTF-8 \ct{LANG} the C
        encoding is used — non-ASCII mangles.
\end{itemize}

\ct{caffeinate -i make} holds it while \ct{make} runs, \ct{-w <pid>} until that pid exits.
\ct{pmset -g log} (sleep/wake history) · \ct{-g therm} · \ct{-g sched}; settings per source:
\ct{-b} battery \ct{-c} charger \ct{-u} UPS \ct{-a} all; \ct{pmset sleepnow}.

\hd{Storage — APFS}
\begin{itemize}
  \item A \textbf{container} shares free space among its \textbf{volumes}. macOS = a
        \textbf{volume group}: System (read-only 10.15, signed + sealed 11) + Data, joined by
        \textbf{firmlinks}.
  \item Snapshot = read-only copy of a volume. \ct{tmutil localsnapshot} ·
        \ct{listlocalsnapshots /} · \ct{deletelocalsnapshots <date>} · \ct{thinlocalsnapshots /};
        \ct{diskutil apfs listSnapshots /}.
  \item \ct{cp -c} = \ct{clonefile(2)}, copy-on-write. \ct{diskutil image create} (26) writes
        \textbf{ASIF}, which replaces \ct{.sparseimage}.
\end{itemize}

\hd{Gatekeeper, quarantine, SIP, TCC}
\begin{itemize}
  \item \ct{spctl -a -vv -t execute App.app} (\ct{install} a \ct{.pkg}, \ct{open} a document):
        exit 0 ok · 1 failed · 2 bad args · \textbf{3 denied}. \ct{\dd global-disable} only
        \emph{reveals} ``Anywhere'' in Privacy \& Security.
  \item \textbf{Quarantine} = xattr \ct{com.apple.quarantine} (not API): Safari sets it, Archive
        Utility propagates it; \ct{curl}/\ct{scp} don't set it, \ct{tar}/\ct{unzip} don't
        propagate. \ct{xattr -d com.apple.quarantine f} · \ct{-r} · \ct{-c} (all).
  \item SIP guards \ct{/System /usr /bin /sbin /var} — not \ct{/usr/local},
        \ct{/Applications}, \ct{/Library}. \ct{csrutil status} (not an API); disable only in
        Recovery.
  \item \ct{tccutil reset AddressBook [bundle-id]} re-prompts; 27: \ct{tccutil list} (managed Macs);
        apps can no longer read the TCC database.
\end{itemize}

\hd{System info, Rosetta, sudo, Homebrew}
\begin{itemize}
  \item \ct{sw\_vers \dd productVersion} · \ct{\dd buildVersion} · \ct{\dd productVersionExtra}
        (Rapid Security Response). \ct{sysctl hw.nperflevels}, \ct{hw.perflevel0.physicalcpu}
        (0 = fastest cores), \ct{hw.memsize}; \ct{sysctl.proc\_translated} 1 = under Rosetta.
  \item \ct{softwareupdate \dd install-rosetta}: Rosetta stays general-purpose through 27, is
        not restored by the upgrade; \ct{sudo nvram boot-args="nox86exec=1"} (26) makes x86
        code crash — proof you no longer need it.
  \item Touch ID \ct{sudo}: \ct{/etc/pam.d/sudo\_local} (14+, from \ct{sudo\_local.template})
        survives updates.
  \item Homebrew: \ct{/opt/homebrew}, no \ct{sudo} after install; macOS 15+, Intel = Tier 3
        (Oct 2026). \ct{brew bundle dump} $\to$ \ct{Brewfile}.
\end{itemize}

\hd{Process inspectors}
\ct{sample <pid>} — all threads' stacks, 10\,s at 1\,ms · \ct{spindump} — the report of a
force-quit hang · \ct{sudo fs\_usage -w -f filesys <proc>} · \ct{footprint} — \emph{dirty}
memory, \ct{phys\_footprint} = kernel ledger · \ct{vmmap \dd summary} · \ct{heap} ·
\ct{leaks \dd atExit \dd{} ./tool}: exit 0 clean, 1 leaks (a CI gate); reads Xcode
\ct{.memgraph} files too.

\hd{Traps}
\begin{itemize}
  \trap{\ct{bootstrap} prints each failing path with its error; \ct{launchctl error 5} decodes
        the number. On 27 a downloaded (quarantined) plist won't load: \ct{xattr -d} it.}
  \trap{\ct{networksetup} wants the \emph{service} (\ct{"Wi-Fi"}), not the device (\ct{en0}).}
  \trap{\ct{leaks}/\ct{vmmap} on a distribution build: no task port\unverified{} — Xcode adds
        \ct{get-task-allow} only when building for debugging.}
\end{itemize}
