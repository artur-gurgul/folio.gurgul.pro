\sheettitle{Privacy compliance for app developers — GDPR, store rules, kids}{security · folio}

\oneliner{The law (\textbf{GDPR}) says: every purpose needs a \textbf{lawful basis}, collect the
\textbf{minimum}, honour \textbf{rights} (access, erasure, portability) and bind every \textbf{processor} by
contract. The stores enforce the visible parts: \textbf{in-app account deletion} (Apple 5.1.1(v), Google
Play), \textbf{Sign in with Apple token revocation}, honest \textbf{privacy labels / Data safety}, \textbf{ATT}
before tracking, stricter rules for \textbf{children}. Your SDKs' data is \emph{your} data.}

\hd{GDPR in one column}
\begin{itemize}
  \item \textbf{Art. 5 principles}: lawful/fair/transparent · purpose limitation · \textbf{data minimisation} ·
        accuracy · storage limitation · integrity \& confidentiality · \textbf{accountability} (prove it).
  \item \textbf{Consent} (Art. 4(11), 7): freely given, specific, informed, unambiguous, an affirmative act;
        withdrawing must be as easy as giving; no pre-ticked boxes, no bundling. Record who, when, what text.
  \item \textbf{ePrivacy} Art. 5(3): reading/writing on the device (SDK identifiers too) needs consent unless
        strictly necessary — that, not only GDPR, is why analytics/ads SDKs wait for opt-in in the EU.
  \item \textbf{Personal data} includes device IDs, IPs, hashed e-mails (pseudonymised $\neq$ anonymous)\unverified.
  \item \textbf{Breach}: tell the authority within \textbf{72 h} (Art. 33). \textbf{DPIA} (Art. 35) before high-risk
        processing (large-scale location, health). Privacy by design/default (Art. 25). Fines up to
        \textbf{€20 M or 4\%} of worldwide turnover.
\end{itemize}

\hd{Example — revoke Sign in with Apple on deletion}
\begin{lstlisting}
POST https://appleid.apple.com/auth/revoke
Content-Type: application/x-www-form-urlencoded

client_id=com.example.app
&client_secret=<JWT: ES256, your .p8 key; iss=Team ID,
                sub=client_id, aud=https://appleid.apple.com>
&token=<refresh_token stored at sign-in>
&token_type_hint=refresh_token
# 200 = revoked (also if already invalid), empty body.
# client_secret: ES256, exp at most 6 months. Also handle
# the server-to-server events consent-revoked, account-deleted
\end{lstlisting}

\hd{Apple guideline 5.1 — the lines to quote}
\begin{itemize}
  \item \textbf{5.1.1(i)} privacy policy linked in App Store Connect \emph{and} in the app; says what, how, retention, how to delete.
  \item \textbf{(ii)} consent even for ``anonymous'' data; paid features can't depend on it; easy withdrawal.
  \item \textbf{(iii)} minimisation: prefer the out-of-process picker or share sheet over full access.
  \item \textbf{(v)} ``If your app supports account creation, you must also offer account deletion within the app'' (since 30 Jun 2022).
  \item \textbf{5.1.2(i)} tracking only with ATT permission; never gate features on it.
\end{itemize}

\hd{ATT is not consent}
ATT (iOS 14.5+) is Apple's gate for \textbf{tracking} (linking with other companies' data) and the IDFA.
GDPR/ePrivacy consent is a separate legal duty with its own UI and records — one does not satisfy the other;
a user can allow ATT and still refuse analytics.

\hd{Children}
\begin{itemize}
  \item \textbf{COPPA} (US, under 13): verifiable parental consent. 2025 amendments — effective 23 Jun 2025,
        comply by \textbf{22 Apr 2026}: \textbf{separate} consent to disclose to third parties (ads), written
        retention policy, no indefinite retention, security programme.
  \item \textbf{GDPR Art. 8}: consent age 16 by default; countries may lower it to 13.
  \item \textbf{Apple Kids category} (1.3, 5.1.4): no third-party analytics/ads (narrow exceptions), parental
        gate. \textbf{Declared Age Range} (iOS 26): the user shares an age \emph{range}, not a birthday —
        for age-assurance laws (US states); Play Age Signals on Android\unverified.
\end{itemize}

\hd{Traps}
\begin{itemize}
  \trap{``The analytics SDK collects it, not us'' — you are the controller; it is on your label and your DPA.}
  \trap{Deletion that leaves the SIWA link, CRM row, analytics profile and logs behind.}
  \trap{Legitimate interest for ad tracking or a consent screen with only ``Accept''.}
  \trap{``We hash e-mails, so it's anonymous'' — still personal data.}
  \trap{Collect ``in case we need it later'' — breaks purpose limitation and minimisation.}
\end{itemize}

\hd{Remember}
\textbf{``A basis per purpose, the minimum, a way out (withdraw, delete) — and every SDK counts as you.''}
