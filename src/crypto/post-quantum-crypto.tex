\sheettitle{Post-quantum cryptography — ML-KEM, ML-DSA, hybrid TLS}{crypto · folio}

\oneliner{A large quantum computer running \textbf{Shor's algorithm} breaks \emph{every}
public-key primitive in use — RSA, (EC)DH, X25519, ECDSA, EdDSA — at any key size; symmetric
ciphers and hashes stand. Traffic recorded \emph{today} decrypts \emph{then} (\textbf{harvest now,
decrypt later}), so \textbf{key exchange} moved first: hybrid \textbf{X25519MLKEM768}, default in
Chrome, Firefox, iOS 26, OpenSSL and OpenSSH. Since the 2026 estimates, \textbf{signatures} are
urgent too: Google and Cloudflare now target 2029 for their whole migration.}

\hd{Why key exchange first — and why hybrid}
\begin{itemize}
  \item Harvested traffic is safe only if its key was agreed post-quantum \emph{at the time}. A
        signature need only hold when checked — but forged firmware and CA signatures work for
        ever: after the 2026 estimates Google ranks authentication first (Android 17: ML-DSA).
  \item Web PKI: no post-quantum X.509 in Chrome's root store — \textbf{Merkle Tree
        Certificates}: a CA signs one tree head, a site sends a short inclusion proof.
  \item \textbf{Hybrid}: secure if \emph{either} part holds; new schemes do fall (SIKE, Rainbow).
        NIST: hybrid is temporary, a second move to PQ-only follows. CNSA 2.0 (NSA) already wants
        pure ML-KEM-1024 + ML-DSA-87, LMS/XMSS for firmware\unverified.
\end{itemize}

  \item \textbf{TLS}: the ClientHello offers \ct{X25519MLKEM768} + a key share; the \emph{server}
        chooses, else classical — nothing to code, the server/CDN must support it. Check with
        \ct{nscurl --tls-diagnostics <url>} (macOS 26). \textbf{Secure Transport} and own TLS
        stacks do not get it.
  \item \textbf{CryptoKit} 26: \ct{MLKEM768}, \ct{MLKEM1024}, \ct{MLDSA65}, \ct{MLDSA87}, their
        \ct{SecureEnclave} variants, HPKE \ct{XWingMLKEM768X25519\_SHA256\_AES\_GCM\_256}; core
        formally verified against FIPS 203; same API server-side in Swift Crypto. 27:
        \ct{MLKEM768.OneTimePrivateKey} — consumed by decapsulate, faster, not storable, not in
        the Secure Enclave.
  \item \textbf{X-Wing} = ML-KEM-768 + X25519 as one KEM: pk 1216, ct 1120, private key a 32-B seed;
        $K$ = SHA3-256($K_M \Vert K_X \Vert ct_X \Vert pk_X \Vert$ label).
\end{itemize}

\hd{Migration — crypto agility}
\begin{enumerate}
  \item \textbf{Inventory} every RSA/ECC use: TLS, VPN, SSH, custom protocols, wrapped keys,
        tokens, signatures, pinned keys, firmware roots (a crypto bill of materials).
  \item Rank by \textbf{secrecy lifetime} (Mosca's $x$) and by \textbf{unpatchable verifiers}.
  \item Ride platform stacks that upgrade themselves (URLSession, OpenSSL, Go) — hand-rolled
        protocols move to HPKE with X-Wing.
  \item Make algorithms \textbf{negotiable}: IDs on the wire, no hard-coded key or signature
        lengths, rotation paths for keys and pins; budget kilobytes for keys, tokens, QR codes.
\end{enumerate}

\hd{Traps}
\begin{itemize}
  \trap{``No quantum computer yet, so no rush'' — today's traffic is the harvest; $x + y > z$.}
  \trap{``Swap ECDH for ML-KEM'' — deploy hybrid; and a bare KEM authenticates nobody.}
  \trap{``Signatures can wait'' — not for roots baked into hardware; estimates keep falling.}
  \trap{Servers and middleboxes that cannot read a large ClientHello — test the fleet.}
  \trap{A bad ciphertext raises no error — implicit rejection just yields a different key.}
\end{itemize}

\hd{Remember}
\textbf{Shor kills public key, Grover barely dents symmetric. Hybrid KEM now, PQ signatures
next, deadlines 2029–2035 — and build for the next swap.}
