\sheettitle{TLS 1.3 in depth — key schedule, records, resumption, ECH}{networking · folio}

\oneliner{TLS 1.3 (RFC 8446) feeds its input secrets — a \textbf{PSK} and the \textbf{ECDHE} result — through
three \textbf{HKDF-Extract} stages (\textbf{Early $\to$ Handshake $\to$ Master}); every key is an
\textbf{HKDF-Expand-Label} of one of them \emph{bound to the transcript hash}, so each phase and direction has
independent keys. Records are AEAD-sealed with a \textbf{per-record nonce}. A \textbf{ticket} makes a PSK for
resumption; \textbf{0-RTT} data is \textbf{replayable}; \textbf{ECH} hides the SNI.}

\hd{How the key schedule works}
\begin{itemize}
  \item \textbf{Extract} $=$ HMAC(salt, input): concentrates entropy. \textbf{Expand-Label} derives keys with a
        \emph{label} + \emph{transcript hash} as context — a key is valid for one role, one direction, one exact
        conversation (tampered hello $\Rightarrow$ other keys $\Rightarrow$ Finished fails).
  \item No ECDHE (\ct{psk\_ke}) $\Rightarrow$ input 0 $\Rightarrow$ \textbf{no forward secrecy}.
        \textbf{Finished} $=$ HMAC(finished\_key, transcript hash).
  \item \ct{SSLKEYLOGFILE} dumps these traffic secrets: enough to decrypt a capture, not to impersonate.
\end{itemize}

\hd{Resumption and 0-RTT (§4.6.1, §8)}
\begin{itemize}
  \item Post-handshake \textbf{NewSessionTicket}: lifetime $\leq$ \textbf{7 days}, \ct{ticket\_age\_add},
        nonce, opaque ticket (usually the PSK sealed under a server \emph{ticket key} — stateless).
  \item Next CH: \ct{pre\_shared\_key} (identity, obfuscated age, \textbf{binder}) + mode \ct{psk\_dhe\_ke}
        (fresh ECDHE, FS) or \ct{psk\_ke}. Saves certificate + signature (\emph{CPU}), still 1 RTT.
  \item \textbf{0-RTT} (\ct{early\_data}): app data right behind the CH, keyed from the PSK alone — a captured
        flight can be \textbf{replayed}. Defences (§8): single-use tickets · ClientHello recording · freshness
        window. HTTP (RFC 8470): \textbf{425 Too Early}; proxies add \ct{Early-Data: 1}.
\end{itemize}

\hd{Example — look inside a real connection}
\begin{lstlisting}[language=bash]
openssl s_client -connect example.com:443 -tls1_3 \
  -servername example.com -alpn h2,http/1.1 \
  -status -sess_out s.pem -keylogfile keys.txt
# -status: print the stapled OCSP answer, if any
openssl s_client -connect example.com:443 \
  -sess_in s.pem -early_data get.txt   # resume + 0-RTT
# keys.txt: CLIENT_TRAFFIC_SECRET_0 ... (for Wireshark)
\end{lstlisting}

\hd{SNI and ECH}
\textbf{SNI} (RFC 6066) is \emph{cleartext}. \textbf{ECH}: RFC 9849, \textbf{March 2026} (+ RFC 9848: config
in the DNS \textbf{HTTPS} record). An \textbf{outer} CH names the provider's \ct{public\_name}; the
\textbf{inner} CH (real SNI, ALPN) is \textbf{HPKE}-encrypted to the provider's key; stale key $\to$
\ct{retry\_configs}; \textbf{GREASE} ECH hides who really uses it. Plain DNS or a dedicated IP still leak.
\textbf{Apple}: \ct{sec\_protocol\_options\_set\_enable\_} \ct{encrypted\_client\_hello} (Network framework) is
documented ``experimental'', default off; no URLSession switch\unverified.

\hd{Revocation — why lives got short}
{\footnotesize
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{14mm}>{\raggedright\arraybackslash}p{58mm}@{}}
\toprule
\textbf{CRL} & signed list of revoked serials; big, stale until reissued \\
\textbf{OCSP} & live query: leaks browsing to the CA, costs an RTT, \textbf{soft-fails} \\
\textbf{stapling} & server attaches a signed OCSP answer (1.3: in the Certificate entry) \\
\textbf{short-lived} & nothing to revoke — the cert dies first \\
\bottomrule
\end{tabular}}\par
Let's Encrypt \textbf{shut OCSP down 6 Aug 2025} (CRLs only); \textbf{6-day certs} GA 15 Jan 2026. CA/B
\textbf{SC-081}: max cert life \textbf{200 d} from 15 Mar 2026 $\to$ \textbf{100 d} (2027) $\to$ \textbf{47 d}
(2029): automate renewal.

\hd{ALPN, QUIC, post-quantum, reuse}
\begin{itemize}
  \item \textbf{ALPN} (RFC 7301): CH offers \ct{h2,http/1.1}; server picks in \textbf{EncryptedExtensions}.
        \ct{h3}: advertised via \ct{Alt-Svc} (RFC 9114), then ALPN \ct{h3} inside QUIC.
  \item \textbf{TLS in QUIC} (RFC 9001): no TLS records — handshake bytes in \textbf{CRYPTO} frames; packet
        keys from the same secrets (\ct{"quic key"}, \ct{"quic iv"}, \ct{"quic hp"}). \textbf{Initial} packets
        are keyed from the Destination Connection ID + a public salt: obfuscated, \emph{not} secret.
  \item \textbf{iOS 26}: URLSession/Network offer hybrid \textbf{X25519MLKEM768} by default; the bigger
        ClientHello breaks some legacy servers that can't read large ones (Apple's warning).
  \item \textbf{Reuse beats tuning}: one long-lived \ct{URLSession} per config (its pool keeps h2/h3 alive),
        never one per request. ARMv8 AES instructions make AES-GCM cheap on phones\unverified.
\end{itemize}

\hd{Traps}
\begin{itemize}
  \trap{``Resumption saves an RTT'' — in 1.3 only \textbf{0-RTT} does, and it is replayable: idempotent requests only.}
  \trap{Stolen \textbf{ticket key} $\to$ every PSK $\to$ all 0-RTT data and \ct{psk\_ke} sessions. Rotate it.}
  \trap{OCSP soft-fail stops no attacker who can block the responder.}
  \trap{\ct{legacy\_session\_id}, \ct{0x0303} in 1.3: middlebox camouflage.}
\end{itemize}

\hd{Remember}
\textbf{``Extract ×3 — Early, Handshake, Master; every key an Expand bound to the transcript.''}
