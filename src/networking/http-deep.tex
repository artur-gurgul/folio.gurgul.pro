\sheettitle{HTTP/1.1 · HTTP/2 · HTTP/3 — and the headers that matter}{networking · folio}

\oneliner{Same \textbf{semantics} (methods, status, headers — RFC 9110) on three \textbf{wire formats}:
1.1 = text, one request at a time per TCP connection; \textbf{2} = binary frames, many
\textbf{streams multiplexed} on one TCP connection (TCP head-of-line blocking remains);
\textbf{3} = the same over \textbf{QUIC/UDP} with TLS 1.3 built in — per-stream loss recovery,
fast setup, \textbf{connection migration}.}

\hd{How it works}
\begin{itemize}
  \item \textbf{1.1} (RFC 9112): connections are \textbf{persistent by default} (\ct{Connection: close}
        ends one) — reuse skips TCP + TLS setup. One response at a time: \textbf{pipelining} had
        to answer in order (one slow response blocks the rest) and proxies broke it — off
        everywhere. Workaround: $\sim$6 connections per host.
  \item \textbf{2} (RFC 9113): frames (\ct{HEADERS DATA SETTINGS WINDOW\_UPDATE RST\_STREAM PING
        GOAWAY}); client streams have odd ids; flow control per stream \emph{and} connection.
        \ct{RST\_STREAM} cancels one request without killing the connection. \textbf{HPACK}: static
        + dynamic table + Huffman — a repeated cookie costs a byte or two. Server push: dead (Chrome
        removed it, 2022\unverified) — use \ct{103 Early Hints}.
  \item \textbf{3} (RFC 9114): streams are independent in QUIC, so loss stalls only its stream.
        \textbf{QPACK} replaces HPACK (HPACK assumes in-order delivery). UDP blocked $\to$ the client
        falls back to TCP.
\end{itemize}

\hd{Caching (RFC 9111) — freshness, then validation}
{\footnotesize
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{25mm}>{\raggedright\arraybackslash}p{47mm}@{}}
\toprule
\ct{max-age=60} & fresh for 60 s — served with no request \\
\ct{no-cache} & \textbf{may store}, must revalidate before every use \\
\ct{no-store} & never write it anywhere (secrets, PII) \\
\ct{private} / \ct{public} & only the end client / shared caches (CDN) too \\
\ct{s-maxage} & freshness for shared caches only \\
\ct{must-revalidate} & no stale reuse after expiry \\
\ct{stale-while-revalidate} & serve stale now, refresh in background \\
\ct{immutable} & never revalidate (hashed asset URLs) \\
\bottomrule
\end{tabular}}\par
\textbf{Validation}: response \ct{ETag: "v7"} (opaque; \ct{W/} = weak) or \ct{Last-Modified} $\to$
request \ct{If-None-Match} / \ct{If-Modified-Since} $\to$ \textbf{304} (no body, reuse cache) or
200. \textbf{\ct{Vary: Accept-Encoding}} adds those request headers to the cache key;
\ct{Vary: *} makes it uncacheable.

\hd{What URLSession does for you}
\begin{itemize}
  \item Negotiates \textbf{h2} via ALPN and \textbf{h3} once the server advertises it — no code.
        Pools and reuses connections \emph{per session}.
  \item Sends \ct{Accept-Encoding: gzip, deflate, br} and decodes transparently; \emph{request}
        bodies are never compressed for you. \ct{expectedContentLength} may be $-1$.
  \item \ct{URLCache} obeys \ct{Cache-Control}; when \emph{it} revalidates, a 304 reaches you as
        200 + the cached body\unverified. Responses over $\sim$5\,\% of the disk cache are not stored.
  \item Cookies: \ct{HTTPCookieStorage} (\ct{httpShouldSetCookies}). \ct{WKWebView} keeps its
        own \ct{WKHTTPCookieStore} — sync them yourself.
\end{itemize}

\hd{CORS — a browser rule}
The \textbf{browser} blocks JS from \emph{reading} a cross-origin response unless the server opts in
(\ct{Access-Control-Allow-Origin}). ``Simple'' requests (GET/HEAD/POST with form types, no custom
headers) go straight out; others — \ct{application/json}, \ct{Authorization}, PUT/DELETE — get an
\ct{OPTIONS} \textbf{preflight} (\ct{Access-Control-Request-Method/-Headers} $\to$
\ct{Allow-Methods/-Headers/-Max-Age}). \ct{Allow-Origin: *} is refused with credentials.
\ct{URLSession} never enforces it; JS inside \ct{WKWebView} does.

\hd{Cookies}
\ct{Secure} (HTTPS only) · \ct{HttpOnly} (no JS — limits XSS theft) · \ct{SameSite=Strict/Lax/None}
(cross-site sending — the CSRF defence; \ct{None} needs \ct{Secure}; Chrome defaults to Lax) ·
prefix \ct{\_\_Host-} $=$ Secure, no Domain, Path=/.

\hd{Traps}
\begin{itemize}
  \trap{\ct{no-cache} is not ``don't cache'' — that is \ct{no-store}.}
  \trap{``HTTP/2 removed head-of-line blocking'' — only at the HTTP layer; TCP still stalls all streams.}
  \trap{Setting \ct{If-None-Match} yourself: now \emph{you} get the raw 304 with an empty body — decode fails.}
  \trap{Per-user response without \ct{private}: a CDN can serve user A's data to user B.}
  \trap{``CORS error in my iOS app'' — impossible in URLSession; and CORS protects users' browsers,
        it is not API auth.}
  \trap{1.1-era domain sharding hurts h2: one connection is the point.}
\end{itemize}

\hd{Remember}
\textbf{``1.1 queues, 2 multiplexes over one pipe, 3 gives every stream its own pipe.''}
