\sheettitle{TypeScript in practice — tsconfig, modules, 6.0 \& 7.0, boundaries}{typescript · folio}

\oneliner{Types are \textbf{erased}: \texttt{tsc} proves consistency \emph{inside} the program,
nothing checks the data that enters it. The config decides how strict the proof is, how imports
resolve for whoever \textbf{loads} the output, and which syntax a per-file transpiler or Node can
strip. Since July 2026 \texttt{tsc} is the native Go port (\textbf{TypeScript 7}); 6.0 was the
last JavaScript-based release.}

\section{Strict, and beyond strict}
\begin{itemize}
  \item \ct{\textbf{strict}} = \ct{alwaysStrict}, \ct{strictNullChecks}, \ct{strictFunctionTypes},
        \ct{strictBindCallApply}, \ct{strictPropertyInitialization}, \ct{noImplicitAny},
        \ct{noImplicitThis}, \ct{useUnknownInCatchVariables} (4.4: \ct{catch (e)} is
        \ct{unknown}), \ct{strictBuiltinIteratorReturn} (5.6: built-in iterators return
        \ct{undefined}, not \ct{any}). New members join on upgrade — pin the TS version.
  \item \textbf{Not in strict}: \ct{noUncheckedIndexedAccess} (4.1, \ct{a[i]}: \ct{T|undefined}),
        \ct{exactOptionalPropertyTypes} (4.4), \ct{noImplicitOverride} (4.3),
        \ct{noPropertyAccessFromIndexSignature} (4.2), \ct{noImplicitReturns},
        \ct{noFallthroughCasesInSwitch}, \ct{noUnusedLocals}/\ct{Parameters}.
  \item \textbf{5.9 \ct{tsc \dd{}init}} turns on the first two above, plus \ct{nodenext},
        \ct{verbatimModuleSyntax}, \ct{isolatedModules}, \ct{moduleDetection: force},
        \ct{skipLibCheck}, \ct{declaration}, \ct{types: []}.
  \item \ct{extends} takes an array (5.0; later wins); \ct{\$\{configDir\}} in path options (5.5)
        = the directory of the config that is compiled, not of the shared base.
  \item \textbf{JS in the program}: \ct{allowJs} mixes \ct{.js} in; \ct{checkJs} or a per-file
        \ct{// @ts-check} checks it from JSDoc, \ct{// @ts-nocheck} opts a file out.
\end{itemize}

\section{Modules — match the loader}
\begin{itemize}
  \item \ct{\textbf{nodenext}}: \ct{.mts} ESM · \ct{.cts} CJS · \ct{.ts} follows the nearest
        \ct{"type"}. ESM relative imports name the \emph{output}: \ct{"./mod.js"}. Default import
        of CJS = \ct{module.exports}. Implies \ct{esModuleInterop} and floating
        \ct{target: esnext}; allows \ct{require(esm)} (5.8, Node 22.12+). Frozen snapshots:
        \ct{node18} (5.8, no \ct{require(esm)}), \ct{node20} (5.9, implies \ct{es2023}).
  \item \ct{\textbf{bundler}} (5.0): extensionless, reads \ct{"exports"} with the \ct{import}
        condition. \ct{\textbf{module: preserve}} (5.4) implies it, plus \ct{esModuleInterop} and
        \ct{resolveJsonModule}; emits \ct{import} as written, \ct{import x = require()} as
        \ct{require}. App recipe: \ct{esnext}+\ct{bundler}, \ct{noEmit},
        \ct{allowImportingTsExtensions} (only with \ct{noEmit} / \ct{emitDeclarationOnly}),
        \ct{verbatimModuleSyntax}; no \ct{"type": "module"} or \ct{.mts} in bundler projects.
  \item \textbf{Library}: \ct{node18}, the \emph{lowest} \ct{target} you support, \ct{strict} (or
        its \ct{.d.ts} breaks strict users), \ct{declaration}+\ct{declarationMap},
        \ct{rootDir}/\ct{outDir} (else users load your \ct{.ts}). \ct{bundler} resolution is
        ``infectious'': it accepts imports Node cannot load.
  \item \ct{"exports"}: TS takes \ct{"types"} first, then \ct{default}/runtime conditions;
        add your own with \ct{customConditions}. \ct{paths} only tells TS — \ct{tsc} never
        rewrites them, so the bundler / Jest / Node (\ct{"imports"}) needs the same map.
\end{itemize}

\section{Enums vs unions}
Numeric \ct{enum} = runtime object + reverse map (\ct{E[0]} is \ct{"A"}); string enums have none.
\ct{const enum} is inlined: ambient ones break \ct{isolatedModules} and inline a stale value when
the library version changes (handbook: lint them away or \ct{preserveConstEnums}). Neither survives
\ct{erasableSyntaxOnly} — use an \ct{as const} object or a literal union.

\section{Erasable TypeScript}
\begin{itemize}
  \item \ct{\textbf{isolatedModules}}: errors where one file is not enough — a type re-exported
        without \ct{export type}, a namespace in a script file, an ambient \ct{const enum}.
  \item \ct{\textbf{verbatimModuleSyntax}} (5.0, below): replaced \ct{importsNotUsedAsValues} and
        \ct{preserveValueImports} (gone in 5.5); CJS output needs \ct{import x = require()}.
  \item \ct{\textbf{erasableSyntaxOnly}} (5.8) bans what has runtime meaning: \ct{enum}, namespaces
        with code, parameter properties, \ct{import =} / \ct{export =}.
  \item \textbf{Node} swaps types for spaces (lines hold, no source map), ignores \ct{tsconfig},
        needs extensions and \ct{import type}; no \ct{.tsx}, no \ct{.ts} in \ct{node\_modules}.
        Its docs: panel 3 + \ct{verbatimModuleSyntax}, \ct{erasableSyntaxOnly},
        \ct{target: esnext}, \ct{noEmit}.
  \item \ct{rewriteRelativeImportExtensions} (5.7): \ct{./a.ts} $\to$ \ct{./a.js} — relative
        literals only; never \ct{paths} aliases, packages, \ct{\#/…}, computed \ct{import()}.
\end{itemize}
\begin{lstlisting}[language=TSSheet]
import type { A } from "a";       // erased entirely
import { b, type c } from "bcd";  // -> import { b } from "bcd";
import { type x } from "xyz";     // -> import {} from "xyz";
\end{lstlisting}

\section{TypeScript 7 — the native compiler}
\begin{itemize}
  \item \ct{npm i -D typescript} = 7.0.2, still \ct{tsc}: VS Code 125.7\,s $\to$ 10.6\,s
        (11.9$\times$), memory $-$6 to $-$26\,\%; \ct{\dd{}checkers N} (4), \ct{\dd{}builders N},
        \ct{\dd{}singleThreaded}.
  \item \textbf{No API in 7.0} (7.1: a new, different one): typed lint rules, webpack loaders,
        Vue / Svelte / Astro / MDX, Angular templates stay on 6 — \ct{@typescript/typescript6}
        ships \ct{tsc6} + the 6.0 API.
  \item Clean on 6.0 with \ct{stableTypeOrdering}, no \ct{ignoreDeprecations} $\Rightarrow$
        should compile identically on 7.0. JSDoc: \ct{@enum} ignored, values not usable as types.
\end{itemize}

\section{References, declarations, speed}
\begin{itemize}
  \item \ct{\textbf{composite}}: \ct{declaration} on, \ct{rootDir} = its dir, every file in
        \ct{include}; importers read its \ct{.d.ts}. \ct{tsc -b} builds stale projects in
        order (\ct{.tsbuildinfo}); since 5.6 past errors (\ct{\dd{}stopOnBuildErrors}).
  \item \ct{\textbf{isolatedDeclarations}} (5.5): exports fully annotated, so any tool emits
        \ct{.d.ts} per file, in parallel; \ct{\dd{}noCheck} (5.6) emits without checking.
  \item \textbf{Cost}: interfaces over big \ct{\&} (cached); annotated returns; unions past a
        dozen members compare pairwise. Measure: \ct{\dd{}extendedDiagnostics},
        \ct{\dd{}explainFiles}, \ct{\dd{}generateTrace} + \ct{@typescript/analyze-trace}.
\end{itemize}

\section{Example — the edges}
\begin{lstlisting}[language=TSSheet]
const User = z.object({ id: z.string(), age: z.number() });
type User = z.infer<typeof User>;   // one source of truth
async function load(res: Response): Promise<User> {
  return User.parse(await res.json()); } // throws ZodError
declare const tag: unique symbol;   // brand: nominal, 0 bytes
type UserId = string & { readonly [tag]: 'UserId' };
const Dir = { Up: 'up', Down: 'down' } as const;  // not enum
type Dir = (typeof Dir)[keyof typeof Dir];        // 'up' | 'down'
// globals.d.ts -- has export: a module, so this augments
export {}; declare global { var __APP_VERSION__: string }
// assets.d.ts -- no import/export: a script, ambient modules
declare module '*.css' {}           // side-effect imports pass
\end{lstlisting}

\section{Traps}
\begin{itemize}
  \trap{6.0 \ct{types: []}: \ct{describe}, \ct{process}, \ct{fs} vanish — list
        \ct{"types": ["node", "jest"]}. \ct{rootDir} = \ct{.}: output lands in \ct{dist/src/}
        unless \ct{"rootDir": "./src"}.}
  \trap{\ct{declare module "x"} in a file that has \emph{any} import/export is an augmentation,
        not a declaration — one stray import turns a shim into a patch.}
  \trap{\ct{@ts-ignore} hides every future error on its line; \ct{@ts-expect-error} (3.9) fails
        with \textit{Unused '@ts-expect-error' directive.} once fixed. typescript-eslint
        \ct{ban-ts-comment}: ignore banned, expect-error needs a $\geq$3-char reason.}
\end{itemize}
